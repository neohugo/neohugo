package hugolib

// Added to package hugolib by the nh-hugolib build oracle (go run -overlay);
// never part of a real build. The oracle also patches (copies of)
// site.go, site_render.go, alias.go and hugo_sites_build.go to call the
// nhBuild* hooks below (see main.go).

import (
	"os"
	"path/filepath"
	"reflect"
	"sort"
	"strings"
	"sync"

	"github.com/neohugo/neohugo/output"
	"github.com/neohugo/neohugo/publisher"
	"github.com/neohugo/neohugo/resources/page"
	"github.com/neohugo/neohugo/resources/postpub"
	"github.com/neohugo/neohugo/tpl/tplimpl"
	"github.com/spf13/afero"
)

// nhBuildRec records one build. The oracle runs Go with
// HUGO_NUMWORKERMULTIPLIER=1, so everything below happens in one goroutine
// at a time; the mutex only guards against surprises.
type nhBuildRec struct {
	mu sync.Mutex
	// "" while rendering, "tpl" while a page template executes
	// (renderForTemplate), "post" from postProcess on.
	phase string
	// File creations in the publish dir, in order.
	events []map[string]any
	// The bytes written to each event's file (same index as events).
	content [][]byte
	// Template executions of renderAndWritePage and publishDestAlias.
	renders []map[string]any
	cur     map[string]any
	// Pages that initialised a paginator (after their render), per format.
	pagers []map[string]any
	// The HTML bytes each site's publisher fed to its elements collector.
	collector map[string][]int
	langs     []string
	// postProcess input.
	toPostProcess []postpub.PostPublishedResource
}

var nhBuild = &nhBuildRec{}

// NHOracleBuildReset starts a new record (before NewHugoSites).
func NHOracleBuildReset() {
	nhBuild = &nhBuildRec{collector: map[string][]int{}}
}

func nhNormPath(name string) string {
	return strings.TrimPrefix(filepath.ToSlash(name), "/")
}

// nhRecFs records every file created (Create, OpenFile with O_CREATE) in
// the publish dir and the bytes written to it.
type nhRecFs struct{ afero.Fs }

// NHNewRecFs wraps the publish dir fs.
func NHNewRecFs(fs afero.Fs) afero.Fs { return nhRecFs{fs} }

type nhRecFile struct {
	afero.File
	idx int
}

func (f nhRecFile) Write(b []byte) (int, error) {
	n, err := f.File.Write(b)
	nhBuild.mu.Lock()
	nhBuild.content[f.idx] = append(nhBuild.content[f.idx], b[:n]...)
	nhBuild.mu.Unlock()
	return n, err
}

func (f nhRecFile) WriteString(s string) (int, error) {
	return f.Write([]byte(s))
}

func (nhBuildRec) created(name string, f afero.File) afero.File {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	idx := len(nhBuild.events)
	nhBuild.events = append(nhBuild.events, map[string]any{"path": nhNormPath(name), "phase": nhBuild.phase})
	nhBuild.content = append(nhBuild.content, []byte{})
	return nhRecFile{File: f, idx: idx}
}

func (fs nhRecFs) Create(name string) (afero.File, error) {
	f, err := fs.Fs.Create(name)
	if err != nil {
		return f, err
	}
	return nhBuild.created(name, f), nil
}

func (fs nhRecFs) OpenFile(name string, flag int, perm os.FileMode) (afero.File, error) {
	f, err := fs.Fs.OpenFile(name, flag, perm)
	if err != nil || flag&os.O_CREATE == 0 {
		return f, err
	}
	return nhBuild.created(name, f), nil
}

// nhPublisher wraps a site's publisher: the HTML bytes it publishes are the
// input of the site's elements collector.
type nhPublisher struct {
	s *Site
	p publisher.Publisher
}

func (w nhPublisher) Publish(d publisher.Descriptor) error {
	nhBuild.mu.Lock()
	before := len(nhBuild.events)
	nhBuild.mu.Unlock()
	err := w.p.Publish(d)
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	if err == nil && len(nhBuild.events) > before {
		idx := len(nhBuild.events) - 1
		if d.StatCounter == &w.s.ProcessingStats.Aliases {
			nhBuild.events[idx]["alias"] = true
		}
		if d.OutputFormat.IsHTML {
			lang := w.s.language.Lang
			nhBuild.collector[lang] = append(nhBuild.collector[lang], idx)
		}
	}
	return err
}

func (w nhPublisher) PublishStats() publisher.PublishStats {
	return w.p.PublishStats()
}

// nhBuildRender records a renderAndWritePage (before its template runs).
func nhBuildRender(s *Site, p *pageState, statCounter *uint64, templ *tplimpl.TemplInfo) {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	kind, n := "page", 0
	if statCounter == &s.ProcessingStats.PaginatorPages {
		kind = "pager"
		n = p.paginator.current.PageNumber()
	}
	name := ""
	if templ != nil {
		name = templ.Name()
	}
	r := map[string]any{
		"kind": kind, "n": n, "lang": p.Lang(), "path": p.Path(), "pkind": p.Kind(),
		"format": p.outputFormat().Name, "templ": name, "empty": true,
	}
	nhBuild.renders = append(nhBuild.renders, r)
	nhBuild.cur = r
}

// nhBuildTemplate marks the template execution of renderForTemplate: files
// created meanwhile (resources published by templates) are tagged "tpl".
func nhBuildTemplate() func(err *error) {
	nhBuild.mu.Lock()
	nhBuild.phase = "tpl"
	nhBuild.mu.Unlock()
	return func(err *error) {
		nhBuild.mu.Lock()
		defer nhBuild.mu.Unlock()
		nhBuild.phase = ""
		if *err != nil && nhBuild.cur != nil {
			nhBuild.cur["err"] = (*err).Error()
		}
	}
}

// nhBuildNonEmpty: the template wrote something (renderAndWritePage
// publishes it).
func nhBuildNonEmpty() {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	if nhBuild.cur != nil {
		nhBuild.cur["empty"] = false
	}
}

// nhBuildPager: p's paginator was initialised when its render finished.
func nhBuildPager(s *Site, p *pageState) {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	nhBuild.pagers = append(nhBuild.pagers, map[string]any{
		"lang": p.Lang(), "path": p.Path(), "format": p.outputFormat().Name,
		"total": p.paginator.current.TotalPages(),
	})
}

// nhBuildAlias records a publishDestAlias.
func nhBuildAlias(s *Site, path, permalink string, f output.Format, p page.Page) {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	lang, pth := "", ""
	if p != nil {
		lang, pth = p.Lang(), p.Path()
	}
	r := map[string]any{
		"kind": "alias", "lang": lang, "path": pth, "format": f.Name,
		"alias": path, "permalink": permalink,
	}
	nhBuild.renders = append(nhBuild.renders, r)
	nhBuild.cur = nil
}

// nhBuildPostProcess is called by postProcess with the resources it
// processes.
func nhBuildPostProcess(rs []postpub.PostPublishedResource) {
	nhBuild.mu.Lock()
	defer nhBuild.mu.Unlock()
	nhBuild.toPostProcess = rs
	nhBuild.phase = "post"
}

// The accessors a PostPublishResource placeholder can name (the ones the
// oracle sites use, plus the MediaType fields).
var nhPostFields = []string{
	"RelPermalink", "Permalink", "Name", "Title", "ResourceType", "Content",
	"MediaType.Type", "MediaType.MainType", "MediaType.SubType", "MediaType.Delimiter",
	"MediaType.SuffixesCSV", "MediaType.String", "MediaType.IsText", "Data.Integrity",
}

// NHOracleBuild builds h (installing the publisher wrappers first) and
// returns the record. enc encodes bytes; norm normalises the site dir.
func NHOracleBuild(h *HugoSites, workingDir string, norm func(string) string, enc func([]byte) any) map[string]any {
	for _, s := range h.Sites {
		s.publisher = nhPublisher{s: s, p: s.publisher}
		nhBuild.langs = append(nhBuild.langs, s.language.Lang)
	}
	buildErr := h.Build(BuildCfg{})

	out := map[string]any{}
	if buildErr != nil {
		out["err"] = norm(buildErr.Error())
	}

	// Events, with the bytes of the alias files.
	events := []any{}
	for i, e := range nhBuild.events {
		ev := map[string]any{"path": e["path"], "phase": e["phase"]}
		if e["alias"] == true {
			ev["bytes"] = enc(nhBuild.content[i])
		}
		events = append(events, ev)
	}
	out["events"] = events
	renders := []any{}
	for _, r := range nhBuild.renders {
		if r["err"] != nil {
			r["err"] = norm(r["err"].(string))
		}
		if r["kind"] == "alias" {
			r["permalink"] = norm(r["permalink"].(string))
		}
		renders = append(renders, r)
	}
	out["renders"] = renders
	pagers := []any{}
	for _, p := range nhBuild.pagers {
		pagers = append(pagers, p)
	}
	out["pagers"] = pagers

	// Build stats: the collector inputs of every site and hugo_stats.json.
	if h.ResourceSpec.BuildConfig().BuildStats.Enabled() {
		inputs := map[string]any{}
		for _, lang := range nhBuild.langs {
			list := []any{}
			for _, idx := range nhBuild.collector[lang] {
				list = append(list, enc(nhBuild.content[idx]))
			}
			inputs[lang] = list
		}
		st := map[string]any{"inputs": inputs, "langs": nhBuild.langs}
		if b, err := os.ReadFile(filepath.Join(workingDir, "hugo_stats.json")); err == nil {
			st["json"] = enc(b)
		}
		out["stats"] = st
	}

	// Post processing: the files with placeholders before and after, and
	// the field values of every PostPublishResource.
	if len(nhBuild.toPostProcess) > 0 {
		var resources []any
		for _, r := range nhBuild.toPostProcess {
			pr := r.(*postpub.PostPublishResource)
			prefix := reflect.ValueOf(pr).Elem().FieldByName("prefix").String()
			fields := map[string]any{}
			for _, f := range nhPostFields {
				v, _ := pr.GetFieldString(prefix + f + postpub.PostProcessSuffix)
				fields[f] = enc([]byte(norm(v)))
			}
			fields["MediaType"] = pr.Origin().MediaType().Type
			resources = append(resources, map[string]any{"prefix": prefix, "fields": fields})
		}
		sort.Slice(resources, func(i, j int) bool {
			return resources[i].(map[string]any)["prefix"].(string) < resources[j].(map[string]any)["prefix"].(string)
		})
		var files []any
		for _, fn := range h.BuildState.GetFilenamesWithPostPrefix() {
			fn = nhNormPath(fn)
			var pre []byte
			found := false
			for i, e := range nhBuild.events {
				if e["path"] == fn && e["phase"] != "post" {
					pre = nhBuild.content[i]
					found = true
				}
			}
			post, err := afero.ReadFile(h.BaseFs.PublishFs, fn)
			f := map[string]any{"path": fn}
			if found {
				f["pre"] = enc(pre)
			}
			if err == nil {
				f["post"] = enc(post)
			}
			files = append(files, f)
		}
		out["postprocess"] = map[string]any{"resources": resources, "files": files}
	}

	// jsconfig.json (written to the project's assets dir when js.Build
	// resolved imports in the project).
	jsRoots := reflect.ValueOf(h.ResourceSpec.JSConfigBuilder).Elem().FieldByName("sourceRoots")
	var roots []string
	for _, k := range jsRoots.MapKeys() {
		roots = append(roots, norm(k.String()))
	}
	sort.Strings(roots)
	if len(roots) > 0 {
		js := map[string]any{"roots": roots}
		if b, err := os.ReadFile(filepath.Join(workingDir, "assets", "jsconfig.json")); err == nil {
			js["json"] = enc(b)
		}
		out["jsconfig"] = js
	}

	return out
}
