var nhCtxType = reflect.TypeOf((*context.Context)(nil)).Elem()
var nhErrType = reflect.TypeOf((*error)(nil)).Elem()

// nhSiteDumper encodes the values page and site methods return, with pages
// as indexes into the page list (walk order of every site, then the pages
// first reached through results).
type nhSiteDumper struct {
	h     *HugoSites
	norm  func(string) string
	enc   func(any) any
	pages []*pageState
	seen  map[*pageState]int
}

func (d *nhSiteDumper) idx(p *pageState) int {
	if i, ok := d.seen[p]; ok {
		return i
	}
	d.seen[p] = len(d.pages)
	d.pages = append(d.pages, p)
	return len(d.pages) - 1
}

func (d *nhSiteDumper) str(s string) any {
	return d.enc(d.norm(s))
}

// value encodes a method result (rv) of declared type t.
func (d *nhSiteDumper) value(rv reflect.Value) any {
	if !rv.IsValid() {
		return map[string]any{"t": "nil"}
	}
	if rv.Kind() == reflect.Interface {
		if rv.IsNil() {
			if rv.Type().NumMethod() == 0 {
				return map[string]any{"t": "nil"}
			}
			return map[string]any{"t": "nil:" + rv.Type().String()}
		}
		rv = rv.Elem()
	}
	return d.any(rv.Interface())
}

func (d *nhSiteDumper) any(v any) any {
	if v == nil {
		return map[string]any{"t": "nil"}
	}
	rv := reflect.ValueOf(v)
	tn := rv.Type().String()
	if (rv.Kind() == reflect.Ptr || rv.Kind() == reflect.Map || rv.Kind() == reflect.Slice) && rv.IsNil() {
		if tn == "*page.nopPage" {
			return map[string]any{"t": tn, "nil": true}
		}
		return map[string]any{"t": "nil:" + tn}
	}
	switch x := v.(type) {
	case *pageState:
		return map[string]any{"t": tn, "p": d.idx(x)}
	case pageWithWeight0:
		return map[string]any{"t": tn, "p": d.idx(x.pageState), "w0": x.weight0}
	case *pageWithOrdinal:
		return map[string]any{"t": tn, "p": d.idx(x.pageState), "ord": x.ordinal}
	case *pageForShortcode:
		return map[string]any{"t": tn, "p": d.idx(x.p)}
	case *pageForRenderHooks:
		return map[string]any{"t": tn, "p": d.idx(x.p)}
	case *pageHeadingsFiltered:
		return map[string]any{"t": tn, "p": d.idx(x.pageState)}
	case page.Page:
		if tn == "*page.nopPage" {
			return map[string]any{"t": tn, "nil": false}
		}
		return map[string]any{"t": tn}
	case *Site:
		return map[string]any{"t": tn, "lang": x.Lang()}
	case page.Site:
		return map[string]any{"t": tn, "lang": x.Language().Lang}
	case page.Pages:
		items := []any{}
		for _, p := range x {
			items = append(items, d.any(p))
		}
		return map[string]any{"t": tn, "items": items}
	case page.Sites:
		items := []any{}
		for _, s := range x {
			items = append(items, d.any(s))
		}
		return map[string]any{"t": tn, "items": items}
	case resource.Resources:
		items := []any{}
		for _, r := range x {
			items = append(items, d.any(r))
		}
		return map[string]any{"t": tn, "items": items}
	case page.OutputFormats:
		items := []any{}
		for _, o := range x {
			items = append(items, d.any(o))
		}
		return map[string]any{"t": tn, "items": items}
	case page.OutputFormat:
		return map[string]any{"t": tn, "name": x.Name(), "rel": x.Rel, "relPermalink": x.RelPermalink(), "permalink": x.Permalink(), "mediaType": x.MediaType().Type}
	case *page.OutputFormat:
		return map[string]any{"t": tn, "name": x.Name(), "rel": x.Rel, "relPermalink": x.RelPermalink(), "permalink": x.Permalink(), "mediaType": x.MediaType().Type}
	case resource.Resource:
		return map[string]any{
			"t": tn, "name": x.Name(), "title": x.Title(), "type": x.ResourceType(),
			"mediaType": x.MediaType().Type, "params": d.enc(x.Params()), "relPermalink": x.RelPermalink(),
		}
	case *langs.Language:
		return map[string]any{"t": tn, "lang": x.Lang}
	case langs.Languages:
		items := []any{}
		for _, l := range x {
			items = append(items, d.any(l))
		}
		return map[string]any{"t": tn, "items": items}
	case *maps.Scratch:
		return map[string]any{"t": tn, "values": d.enc(x.Values())}
	case config.SitemapConfig:
		return map[string]any{"t": tn, "changeFreq": x.ChangeFreq, "priority": d.enc(x.Priority), "filename": x.Filename, "disable": x.Disable}
	case page.SiteConfig:
		return map[string]any{"t": tn, "rssLimit": x.Services.RSS.Limit}
	case media.Type:
		return map[string]any{"t": tn, "type": x.Type}
	case neohugo.HugoInfo:
		return map[string]any{"t": tn, "environment": x.Environment}
	case *source.File:
		return map[string]any{"t": tn, "path": x.Path(), "uniqueID": x.UniqueID(), "lang": x.Lang(), "section": x.Section()}
	case page.Data:
		return d.mapEntries(tn, reflect.ValueOf(x))
	case page.Taxonomy, page.TaxonomyList, navigation.Menus, navigation.PageMenus, page.AuthorList:
		return d.mapEntries(tn, reflect.ValueOf(x))
	case page.WeightedPages:
		items := []any{}
		for _, w := range x {
			items = append(items, map[string]any{"w": w.Weight, "p": d.any(w.Page)})
		}
		return map[string]any{"t": tn, "items": items}
	case navigation.Menu:
		items := []any{}
		for _, e := range x {
			items = append(items, d.any(e))
		}
		return map[string]any{"t": tn, "items": items}
	case *navigation.MenuEntry:
		var pv any
		if x.Page != nil {
			pv = d.any(x.Page)
		}
		return map[string]any{
			"t": tn, "identifier": x.Identifier, "parent": x.Parent, "name": x.Name, "pre": string(x.Pre), "post": string(x.Post),
			"url": x.URL(), "configuredURL": x.ConfiguredURL, "pageRef": x.PageRef, "weight": x.Weight, "title": x.Title,
			"menu": x.Menu, "params": d.enc(x.Params), "page": pv, "children": d.any(x.Children), "hasChildren": x.HasChildren(), "keyName": x.KeyName(),
		}
	case *page.Pager:
		return map[string]any{
			"t": tn, "number": x.PageNumber(), "url": x.URL(), "totalPages": x.TotalPages(), "pagerSize": x.PagerSize(),
			"numberOfElements": x.NumberOfElements(), "totalNumberOfElements": x.TotalNumberOfElements(),
			"hasNext": x.HasNext(), "hasPrev": x.HasPrev(), "pages": d.any(x.Pages()),
		}
	case string:
		return d.str(x)
	case template.HTML:
		return d.enc(template.HTML(d.norm(string(x))))
	case time.Time, bool, int, int64, uint32, float64, []string, maps.Params, map[string]any, map[string]string:
		return d.enc(x)
	}
	if rv.Kind() == reflect.Func {
		return map[string]any{"t": tn}
	}
	return map[string]any{"t": tn, "unencoded": true}
}

func (d *nhSiteDumper) mapEntries(tn string, rv reflect.Value) any {
	var keys []string
	for _, k := range rv.MapKeys() {
		keys = append(keys, k.String())
	}
	sort.Strings(keys)
	entries := []any{}
	for _, k := range keys {
		entries = append(entries, []any{k, d.any(rv.MapIndex(reflect.ValueOf(k).Convert(rv.Type().Key())).Interface())})
	}
	return map[string]any{"t": tn, "entries": entries}
}

// call calls method name of recv with args (a context is prepended when the
// method takes one) and encodes the result: {"ok": v}, {"err": msg} or
// {"panic": msg}; methods without a result give {"ok0": true}.
func (d *nhSiteDumper) call(recv any, name string, args ...any) (res any) {
	m := reflect.ValueOf(recv).MethodByName(name)
	if !m.IsValid() {
		return map[string]any{"missing": true}
	}
	mt := m.Type()
	var in []reflect.Value
	if mt.NumIn() > 0 && mt.In(0) == nhCtxType {
		in = append(in, reflect.ValueOf(context.Background()))
	}
	for i, a := range args {
		pi := len(in)
		var at reflect.Type
		if mt.IsVariadic() && pi >= mt.NumIn()-1 {
			at = mt.In(mt.NumIn() - 1).Elem()
		} else {
			at = mt.In(pi)
		}
		if a == nil {
			in = append(in, reflect.Zero(at))
		} else {
			v := reflect.ValueOf(a)
			if !v.Type().AssignableTo(at) {
				return map[string]any{"badarg": i}
			}
			in = append(in, v)
		}
	}
	defer func() {
		if r := recover(); r != nil {
			res = map[string]any{"panic": d.norm(fmt.Sprint(r))}
		}
	}()
	out := m.Call(in)
	switch len(out) {
	case 0:
		return map[string]any{"ok0": true}
	case 2:
		if mt.Out(1) == nhErrType && !out[1].IsNil() {
			return map[string]any{"err": d.norm(out[1].Interface().(error).Error())}
		}
	}
	return map[string]any{"ok": d.value(out[0])}
}

// NHOracleSite runs process and assemble, then calls the template-visible
// methods of every page and site (the tplapi tables' Go counterparts) and
// records the results, the paginators, and the output of the probe
// templates (no template functions: the Rust test runs them with a
// names-only func map).
// nhCanonicalMainSections makes the recorded main section reproducible: without
// configured main sections, applyAggregates picks the section with the most
// regular pages by ranging over a map, so with a tie Go picks any of the tied
// sections. The oracle then records the first in byte order (one of Go's
// results, and the port's choice).
func nhCanonicalMainSections(h *HugoSites, configured []bool) error {
	for i, s := range h.Sites {
		ms := s.conf.C.MainSections
		if configured[i] || len(ms) != 1 {
			continue
		}
		counts := map[string]int{}
		w := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: s.pageMap.treePages,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				if p, ok := n.(*pageState); ok && p.IsPage() {
					counts[p.Section()]++
				}
				return false, nil
			},
		}
		if err := w.Walk(context.Background()); err != nil {
			return err
		}
		top := counts[ms[0]]
		var tied []string
		for sec, c := range counts {
			if sec != "" && c == top {
				tied = append(tied, sec)
			}
		}
		sort.Strings(tied)
		if top > 0 && len(tied) > 1 && tied[0] != ms[0] {
			s.conf.C.SetMainSections([]string{tied[0]})
		}
	}
	return nil
}

func NHOracleSite(h *HugoSites, norm func(string) string, enc func(any) any, probes [][2]string) (map[string]any, error) {
	conf := &BuildCfg{WhatChanged: &WhatChanged{needsPagesAssembly: true}}
	l := h.Log.InfoCommand("build")
	var configuredMainSections []bool
	for _, s := range h.Sites {
		configuredMainSections = append(configuredMainSections, s.conf.C.IsMainSectionsSet())
	}
	if err := h.process(context.Background(), l, conf, func(*BuildCfg) error { return nil }); err != nil {
		return nil, err
	}
	if err := h.assemble(context.Background(), l, conf); err != nil {
		return nil, err
	}
	if err := nhCanonicalMainSections(h, configuredMainSections); err != nil {
		return nil, err
	}
	for _, s := range h.Sites {
		s.state = siteStateReady
	}

	d := &nhSiteDumper{h: h, norm: norm, enc: enc, seen: map[*pageState]int{}}

	// The page list: every site's pages in walk order, then its bundled pages.
	var siteNodes [][]*pageState
	for _, s := range h.Sites {
		var nodes []*pageState
		w := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: s.pageMap.treePages,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				p := n.(*pageState)
				d.idx(p)
				nodes = append(nodes, p)
				return false, nil
			},
		}
		if err := w.Walk(context.Background()); err != nil {
			return nil, err
		}
		rw := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: s.pageMap.treeResources,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				if rs, ok := n.(*resourceSource); ok {
					if p, ok := rs.r.(*pageState); ok {
						d.idx(p)
						nodes = append(nodes, p)
					}
				}
				return false, nil
			},
		}
		if err := rw.Walk(context.Background()); err != nil {
			return nil, err
		}
		siteNodes = append(siteNodes, nodes)
	}

	nop := page.NilPage

	// Sites.
	siteDumps := []any{}
	for si, s := range h.Sites {
		sw := s.Site()
		sd := map[string]any{"lang": s.Lang()}
		calls := []any{}
		for _, name := range []string{
			"AllPages", "Author", "Authors", "BaseURL", "BuildDrafts", "Config", "Copyright", "Current", "Data",
			"Home", "Hugo", "IsMultiLingual", "Key", "Language", "LanguageCode", "LanguagePrefix", "Languages",
			"LastChange", "Lastmod", "MainSections", "Menus", "Pages", "Params", "RegularPages", "Sections",
			"ServerPort", "Sites", "Social", "Store", "Taxonomies", "Title",
		} {
			calls = append(calls, []any{name, d.call(sw, name)})
		}
		refs := []string{"", "/", "home", "section", "/nope", "404", "sitemap", "/tags", "tags"}
		for _, p := range siteNodes[si] {
			refs = append(refs, p.Path())
		}
		for _, r := range refs {
			calls = append(calls, []any{"GetPage", r, d.call(sw, "GetPage", r)})
		}
		calls = append(calls, []any{"GetPage", "section|blog", d.call(sw, "GetPage", "section", "blog")})
		calls = append(calls, []any{"GetPage", "a|b|c", d.call(sw, "GetPage", "a", "b", "c")})
		var pkeys []string
		for k := range s.Params() {
			pkeys = append(pkeys, k)
		}
		sort.Strings(pkeys)
		pkeys = append(pkeys, "nope", "MAINSECTIONS", "a.b")
		for _, k := range pkeys {
			calls = append(calls, []any{"Param", k, d.call(sw, "Param", k)})
		}
		sd["siteWrapper"] = calls

		// *hugolib.Site (the sitemapindex data).
		hcalls := []any{}
		for _, name := range []string{"SitemapAbsURL", "Lastmod", "BaseURL", "Title", "GetLanguagePrefix", "GetTargetLanguageBasePath", "Lang", "LanguagePrefix", "ServerPort", "Site", "Current", "MainSections"} {
			hcalls = append(hcalls, []any{name, d.call(s, name)})
		}
		for _, a := range [][]any{
			{"AbsURL", "a/b.html", true}, {"AbsURL", "/a/b/", false}, {"AbsURL", "https://x.org/y", false},
			{"RelURL", "a/b.html", true}, {"RelURL", "/a/b/", false},
			{"URLize", "Hello World/Ünïcode & Co"}, {"URLizeFilename", "a b/c.md"}, {"URLEscape", "a b?c=d#e"},
			{"MakePath", "A B/c.md"}, {"MakePathSanitized", "A B/c.md"},
			{"PrependBasePath", "/a/b", false}, {"PrependBasePath", "/a/b", true},
			{"PermalinkForBaseURL", "/x/", "https://b.org/sub/"}, {"IsAbsURL", "https://x.org"}, {"IsAbsURL", "/x"},
			{"GetBasePath", true}, {"GetBasePath", false}, {"SanitizeAnchorName", "Hello World ÄÖ"}, {"ResolveMarkup", "md"},
		} {
			hcalls = append(hcalls, []any{a[0], a[1:], d.call(s, a[0].(string), a[1:]...)})
		}
		sd["hugolibSite"] = hcalls

		sd["paramsSame"] = reflect.ValueOf(sw.Params()).Pointer() == reflect.ValueOf(sw.Params()).Pointer() &&
			reflect.ValueOf(sw.Params()).Pointer() == reflect.ValueOf(s.Language().Params()).Pointer()
		siteDumps = append(siteDumps, sd)
	}

	pageArgs := func(p *pageState, si int) [][]any {
		s := h.Sites[si]
		var out [][]any
		home := s.home
		var others []any
		others = append(others, p, nop, "str", nil)
		if home != nil {
			others = append(others, home)
		}
		if pp := p.Parent(); pp != nil {
			others = append(others, pp)
		}
		for _, o := range others {
			out = append(out, []any{"Eq", o}, []any{"InSection", o}, []any{"IsAncestor", o}, []any{"IsDescendant", o})
		}
		for _, r := range []string{"", ".", "..", "/", "nope", "/nope", "index.md", "_index.md", "../_index.md"} {
			out = append(out, []any{"GetPage", r})
		}
		for _, v := range s.pageMap.cfg.taxonomyConfig.views {
			out = append(out, []any{"GetTerms", v.plural})
		}
		out = append(out, []any{"GetTerms", "nope"})
		for _, n := range []string{"ref", "relref", "badge", "note", "box", "nope"} {
			out = append(out, []any{"HasShortcode", n})
		}
		var pkeys []string
		for k := range p.Params() {
			pkeys = append(pkeys, k)
		}
		sort.Strings(pkeys)
		if len(pkeys) > 6 {
			pkeys = pkeys[:6]
		}
		pkeys = append(pkeys, "Title", "nope", "a.b")
		for _, k := range pkeys {
			out = append(out, []any{"Param", k})
		}
		var targets []string
		targets = append(targets, p.Path(), "/", "nope", "/nope", "", "#frag", p.Path()+"#frag", "..")
		if home != nil {
			targets = append(targets, home.Path()+"#top")
		}
		var langs []string
		for _, ss := range h.Sites {
			langs = append(langs, ss.Lang())
		}
		for _, t := range targets {
			m := map[string]any{"path": t}
			out = append(out, []any{"Ref", m}, []any{"RelRef", m}, []any{"RefFrom", m, p}, []any{"RelRefFrom", m, p})
		}
		for _, lang := range append(langs, "xx") {
			m := map[string]any{"path": "/", "lang": lang}
			out = append(out, []any{"Ref", m}, []any{"RelRef", m})
		}
		for _, of := range []string{"json", "rss", "html", "nope"} {
			m := map[string]any{"path": "/", "outputFormat": of}
			out = append(out, []any{"Ref", m}, []any{"RelRef", m})
		}
		out = append(out, []any{"Ref", map[string]any{"Path": p.Path(), "LANG": s.Lang()}})
		out = append(out, []any{"Ref", map[string]any{"path": 42}})
		out = append(out, []any{"Ref", map[string]any{"path": []string{"a"}}})
		out = append(out, []any{"Ref", map[string]any{}})
		out = append(out, []any{"Ref", map[string]any(nil)})
		// Menus of the site.
		var mnames []string
		for k := range s.Menus() {
			mnames = append(mnames, k)
		}
		sort.Strings(mnames)
		for _, mn := range mnames {
			for i, me := range s.Menus()[mn] {
				if i >= 3 {
					break
				}
				out = append(out, []any{"HasMenuCurrent", mn, me}, []any{"IsMenuCurrent", mn, me})
				for j, c := range me.Children {
					if j >= 2 {
						break
					}
					out = append(out, []any{"HasMenuCurrent", mn, c}, []any{"IsMenuCurrent", mn, c})
				}
			}
		}
		return out
	}

	zeroArg := []string{
		"Aliases", "AllTranslations", "AlternativeOutputFormats", "Ancestors", "BundleType", "CodeOwners", "CurrentSection",
		"Data", "Date", "Description", "Draft", "ExpiryDate", "File", "FirstSection", "GitInfo", "IdentifierBase", "IsHome",
		"IsNode", "IsPage", "IsSection", "IsTranslated", "Keywords", "Kind", "Lang", "Language", "Lastmod", "Layout",
		"LinkTitle", "MediaType", "Menus", "Name", "Next", "NextInSection", "NextPage", "OutputFormats", "Page", "Pages",
		"PagesRecursive", "Params", "Parent", "Path", "Permalink", "Prev", "PrevInSection", "PrevPage", "PublishDate",
		"RawContent", "RegularPages", "RegularPagesRecursive", "RelPermalink", "ResourceType", "Resources", "Section",
		"Sections", "SectionsEntries", "SectionsPath", "Site", "Sitemap", "Sites", "Slug", "StaleVersion", "String", "Title",
		"TranslationKey", "Translations", "Type", "Weight", "HeadingsFiltered", "Weight0", "Ordinal", "Unwrapv",
	}
	wrapperOnly := []string{
		"Content", "ContentWithoutSummary", "Plain", "PlainWords", "Summary", "Truncated", "FuzzyWordCount", "WordCount",
		"ReadingTime", "Len", "Markup",
	}

	// Big sites (docs/) dump every 10th page; arguments are tried on every page of
	// small sites and every 4th page of the others (fixture size).
	dumpPage := func(si, wi int) bool { return len(siteNodes[si]) <= 200 || wi%10 == 0 }
	argPage := func(si, wi int) bool { return len(siteNodes[si]) <= 60 || wi%4 == 0 }

	pageCalls := func(recv any, p *pageState, si int, wrapper string, withArgs bool) []any {
		calls := []any{}
		for _, name := range zeroArg {
			if reflect.ValueOf(recv).MethodByName(name).IsValid() {
				calls = append(calls, []any{name, nil, d.call(recv, name)})
			}
		}
		if wrapper != "" {
			for _, name := range wrapperOnly {
				calls = append(calls, []any{name, nil, d.call(recv, name)})
			}
			if wrapper == "shortcode" {
				calls = append(calls, []any{"TableOfContents", nil, d.call(recv, "TableOfContents")})
			}
		}
		var args [][]any
		if withArgs {
			args = pageArgs(p, si)
		}
		for _, a := range args {
			name := a[0].(string)
			var encArgs []any
			for _, x := range a[1:] {
				encArgs = append(encArgs, d.any(x))
			}
			calls = append(calls, []any{name, encArgs, d.call(recv, name, a[1:]...)})
		}
		// The page's store is shared by .Scratch and .Store.
		if sc, ok := d.call(recv, "Scratch").(map[string]any)["ok"]; ok && sc != nil {
			scratch := reflect.ValueOf(recv).MethodByName("Scratch").Call(nil)[0].Interface().(*maps.Scratch)
			scratch.Set("probe-"+wrapper, d.idx(p))
			calls = append(calls, []any{"Store", nil, d.call(recv, "Store")})
		}
		return calls
	}

	// Pages: the method calls in walk order (the position, menu and paginator
	// inits run on first use, as in a build).
	pageDumps := []any{}
	for si := range h.Sites {
		for wi, p := range siteNodes[si] {
			if !dumpPage(si, wi) {
				continue
			}
			withArgs := argPage(si, wi)
			pd := map[string]any{"p": d.idx(p), "site": si}
			pd["calls"] = pageCalls(p, p, si, "", withArgs)
			// The wrappers of every 5th page.
			if wi%5 == 0 {
				pd["shortcode"] = pageCalls(newPageForShortcode(p), p, si, "shortcode", false)
				pd["renderHooks"] = pageCalls(newPageForRenderHook(p), p, si, "renderhooks", false)
			}
			// Weight0 / Ordinal wrappers come from the collections.
			for _, t := range p.Pages() {
				if w, ok := t.(pageWithWeight0); ok {
					pd["weight0"] = pageCalls(w, w.pageState, si, "", false)
					break
				}
			}
			for _, v := range h.Sites[si].pageMap.cfg.taxonomyConfig.views {
				ts := p.GetTerms(v.plural)
				if len(ts) > 0 {
					pd["ordinal"] = pageCalls(ts[0], ts[0].(*pageWithOrdinal).pageState, si, "", false)
					break
				}
			}
			pageDumps = append(pageDumps, pd)
		}
	}

	// Paginators: every node; every other node calls .Paginate (size 2) first,
	// which then wins over .Paginator.
	pagerDump := func(pg *page.Pager, err error) any {
		if err != nil {
			return map[string]any{"err": d.norm(err.Error())}
		}
		if pg == nil {
			return map[string]any{"nil": true}
		}
		var pagers []any
		for _, pp := range pg.Pagers() {
			pagers = append(pagers, d.any(pp))
		}
		return map[string]any{"current": d.any(pg), "pagers": pagers, "first": pg.First().PageNumber(), "last": pg.Last().PageNumber()}
	}
	paginators := []any{}
	for si := range h.Sites {
		for wi, p := range siteNodes[si] {
			if !p.IsNode() || !dumpPage(si, wi) {
				continue
			}
			e := map[string]any{"p": d.idx(p)}
			if wi%2 == 1 {
				pg, err := p.Paginate(p.Pages(), 2)
				e["paginate"] = pagerDump(pg, err)
			}
			pg, err := p.Paginator()
			e["paginator"] = pagerDump(pg, err)
			pg, err = p.Paginate(p.RegularPages(), 1)
			e["paginateAfter"] = pagerDump(pg, err)
			paginators = append(paginators, e)
		}
	}
	// A rendering-site shift resets the built paginators: the next call builds
	// again (here with an explicit size).
	for si, s := range h.Sites {
		if err := s.preparePagesForRender(true, 0); err != nil {
			return nil, err
		}
		for wi, p := range siteNodes[si] {
			if !p.IsNode() || !dumpPage(si, wi) {
				continue
			}
			pg, err := p.Paginator(3)
			paginators = append(paginators, map[string]any{"p": d.idx(p), "afterReset": pagerDump(pg, err)})
		}
	}

	// Probe templates (after the shifts: the first output of every page).
	probeDumps := []any{}
	for si, s := range h.Sites {
		for wi, p := range siteNodes[si] {
			if (wi%3 != 0 && !p.IsHome()) || !dumpPage(si, wi) {
				continue
			}
			for _, pr := range probes {
				ti, err := s.GetTemplateStore().TextParse("probe-"+pr[0], pr[1])
				if err != nil {
					return nil, err
				}
				var buf bytes.Buffer
				ctx := tpl.Context.Page.Set(context.Background(), p)
				err = s.GetTemplateStore().ExecuteWithContext(ctx, ti, &buf, p)
				e := map[string]any{"p": d.idx(p), "probe": pr[0], "out": d.norm(buf.String())}
				if err != nil {
					e["err"] = d.norm(err.Error())
				}
				probeDumps = append(probeDumps, e)
			}
		}
	}

	// The page list, with what identifies a page.
	list := []any{}
	for i := 0; i < len(d.pages); i++ {
		p := d.pages[i]
		list = append(list, map[string]any{"site": p.s.languagei, "kind": p.Kind(), "path": p.Path(), "lang": p.Lang()})
	}

	_ = kinds.KindPage
	_ = strings.TrimSpace
	return map[string]any{
		"sites":      siteDumps,
		"pages":      pageDumps,
		"paginators": paginators,
		"probes":     probeDumps,
		"pageList":   list,
	}, nil
}

// NHOracleMethodSets returns the exported method sets of the types templates
// receive, from reflect (name, parameter count without the receiver, variadic,
// a context.Context first parameter, result count, whether text/template can
// call it (goodFunc), parameter and result types).
func NHOracleMethodSets(h *HugoSites) []any {
	s := h.Sites[0]
	types := []reflect.Type{
		reflect.TypeOf((*pageState)(nil)),
		reflect.TypeOf(pageWithWeight0{}),
		reflect.TypeOf((*pageWithOrdinal)(nil)),
		reflect.TypeOf((*pageForShortcode)(nil)),
		reflect.TypeOf((*pageForRenderHooks)(nil)),
		reflect.TypeOf((*pageHeadingsFiltered)(nil)),
		reflect.TypeOf(page.NilPage),
		reflect.TypeOf(s),
		reflect.TypeOf(s.Site()),
		reflect.TypeOf(page.Pages{}),
		reflect.TypeOf(page.PageGroup{}),
		reflect.TypeOf(page.PagesGroup{}),
		reflect.TypeOf((*page.Pager)(nil)),
		reflect.TypeOf(page.WeightedPages{}),
		reflect.TypeOf(page.WeightedPage{}),
		reflect.TypeOf(page.Taxonomy{}),
		reflect.TypeOf(page.TaxonomyList{}),
		reflect.TypeOf(page.OrderedTaxonomy{}),
		reflect.TypeOf(page.OrderedTaxonomyEntry{}),
		reflect.TypeOf(page.Data{}),
		reflect.TypeOf(page.OutputFormats{}),
		reflect.TypeOf(page.OutputFormat{}),
		reflect.TypeOf(&page.OutputFormat{}),
		reflect.TypeOf(resource.Resources{}),
		reflect.TypeOf(maps.Params{}),
		reflect.TypeOf(&maps.Scratch{}),
		reflect.TypeOf(navigation.Menu{}),
		reflect.TypeOf(navigation.Menus{}),
		reflect.TypeOf(navigation.PageMenus{}),
		reflect.TypeOf(&navigation.MenuEntry{}),
		reflect.TypeOf(langs.Languages{}),
		reflect.TypeOf(&langs.Language{}),
		reflect.TypeOf(config.SitemapConfig{}),
		reflect.TypeOf(page.SiteConfig{}),
		reflect.TypeOf(&source.File{}),
		reflect.TypeOf(s.Hugo()),
		reflect.TypeOf(media.Type{}),
		reflect.TypeOf(page.Sites{}),
	}
	pager := reflect.TypeOf((*page.Pager)(nil))
	if m, ok := pager.MethodByName("Pagers"); ok {
		types = append(types, m.Type.Out(0))
	}
	out := []any{}
	for _, t := range types {
		ms := []any{}
		for i := 0; i < t.NumMethod(); i++ {
			m := t.Method(i)
			mt := m.Type
			var in []string
			for j := 1; j < mt.NumIn(); j++ {
				in = append(in, mt.In(j).String())
			}
			var o []string
			for j := 0; j < mt.NumOut(); j++ {
				o = append(o, mt.Out(j).String())
			}
			good := mt.NumOut() == 1 || (mt.NumOut() == 2 && mt.Out(1) == nhErrType)
			ms = append(ms, map[string]any{
				"name": m.Name, "numIn": mt.NumIn() - 1, "variadic": mt.IsVariadic(),
				"ctx": mt.NumIn() > 1 && mt.In(1) == nhCtxType, "numOut": mt.NumOut(), "good": good,
				"in": in, "out": o,
			})
		}
		// Exported struct fields a template can read (Go evalField after the methods).
		var fields []string
		st := t
		if st.Kind() == reflect.Ptr {
			st = st.Elem()
		}
		if st.Kind() == reflect.Struct {
			for _, f := range reflect.VisibleFields(st) {
				if f.IsExported() {
					if _, ok := t.MethodByName(f.Name); !ok {
						fields = append(fields, f.Name)
					}
				}
			}
			sort.Strings(fields)
		}
		out = append(out, map[string]any{"type": t.String(), "kind": t.Kind().String(), "methods": ms, "fields": fields})
	}
	return out
}
