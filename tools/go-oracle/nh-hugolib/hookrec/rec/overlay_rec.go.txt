// nhExecRecord is one render-hook or shortcode template execution.
type nhExecRecord struct {
	p       *pageState
	format  string
	kind    string
	ordinal int
	out     string
	err     string
	nested  bool
	// The template context's IsInGoldmark (set only for {{% %}} shortcodes).
	inGoldmark bool
}

// nhRec is the recorder. Builds run with one worker, so a plain global
// suffices. Executions started while another recorded execution runs
// (e.g. hooks of a markdownify call inside a shortcode template) are nested:
// the outer output already contains them.
var nhRec struct {
	on      bool
	depth   int
	counts  map[string]int
	records []nhExecRecord
}

func nhRecordExec(ctx context.Context, p *pageState, format, kind string, w io.Writer, exec func(io.Writer) error) error {
	if !nhRec.on {
		return exec(w)
	}
	r := nhExecRecord{p: p, format: format, kind: kind, nested: nhRec.depth > 0, inGoldmark: tpl.Context.IsInGoldmark.Get(ctx)}
	if !r.nested {
		key := fmt.Sprintf("%p|%s|%s", p, format, kind)
		r.ordinal = nhRec.counts[key]
		nhRec.counts[key]++
	}
	idx := len(nhRec.records)
	nhRec.records = append(nhRec.records, r)
	nhRec.depth++
	var buf bytes.Buffer
	err := exec(&buf)
	nhRec.depth--
	nhRec.records[idx].out = buf.String()
	if err != nil {
		nhRec.records[idx].err = err.Error()
		return err
	}
	_, err = w.Write(buf.Bytes())
	return err
}

func nhHook(hr hookRendererTemplate, cctx context.Context, w io.Writer, exec func(io.Writer) error) error {
	return nhRecordExec(cctx, hr.nhPco.po.p, hr.nhPco.po.f.Name, hr.nhKind, w, exec)
}

func nhShortcode(ctx context.Context, data *ShortcodeWithPage, buf *bytes.Buffer, exec func(*bytes.Buffer) error) error {
	p := data.Page.(*pageForShortcode).p
	return nhRecordExec(ctx, p, data.nhFormat, "shortcode:"+data.Name, buf, func(w io.Writer) error {
		return exec(w.(*bytes.Buffer))
	})
}

// NHRecStart starts recording.
func NHRecStart() {
	nhRec.on = true
	nhRec.depth = 0
	nhRec.counts = map[string]int{}
	nhRec.records = nil
}

// NHRecStop stops recording.
func NHRecStop() {
	nhRec.on = false
}

// nhPages lists the pages with a file of every site (forEeachPageIncludingBundledPages order).
func nhPages(h *HugoSites) ([]*pageState, map[*pageState]int, error) {
	var pages []*pageState
	index := map[*pageState]int{}
	for _, s := range h.Sites {
		if err := s.pageMap.forEeachPageIncludingBundledPages(nil, func(p *pageState) (bool, error) {
			if p.File() == nil {
				return false, nil
			}
			if _, ok := index[p]; ok {
				return false, nil
			}
			index[p] = len(pages)
			pages = append(pages, p)
			return false, nil
		}); err != nil {
			return nil, nil, err
		}
	}
	return pages, index, nil
}

func nhPageDump(p *pageState, norm func(string) string) map[string]any {
	pc := p.m.pageConfig
	return map[string]any{
		"file":       norm(p.File().Filename()),
		"lang":       p.s.Lang(),
		"site":       p.s.languagei,
		"kind":       p.Kind(),
		"path":       p.Path(),
		"type":       pc.Type,
		"layout":     pc.Layout,
		"markup":     pc.Content.Markup,
		"mediaType":  pc.ContentMediaType.Type,
		"summary":    pc.Summary,
		"isCJK":      pc.IsCJKLanguage,
		"title":      pc.Title,
		"standalone": !p.m.standaloneOutputFormat.IsZero(),
		"pid":        p.pid,
	}
}

func nhRecordsDump(index map[*pageState]int) []any {
	recs := []any{}
	for _, r := range nhRec.records {
		m := map[string]any{
			"format":  r.format,
			"kind":    r.kind,
			"ordinal": r.ordinal,
			"out":     r.out,
			"nested":  r.nested,
			"inGoldmark": r.inGoldmark,
			"lang":    r.p.s.Lang(),
			"path":    r.p.Path(),
		}
		if i, ok := index[r.p]; ok {
			m["page"] = i
		} else {
			m["page"] = -1
		}
		if r.err != "" {
			m["err"] = r.err
		}
		recs = append(recs, m)
	}
	return recs
}

func nhFormats(h *HugoSites) ([]any, []any) {
	global := []any{}
	for _, f := range h.renderFormats {
		global = append(global, f.Name)
	}
	sites := []any{}
	for _, s := range h.Sites {
		names := []any{}
		for _, f := range s.renderFormats {
			names = append(names, f.Name)
		}
		sites = append(sites, map[string]any{"lang": s.Lang(), "renderFormats": names})
	}
	return global, sites
}

func nhErr(err error, norm func(string) string) any {
	if err == nil {
		return nil
	}
	return norm(err.Error())
}

// nhContentValues calls the content methods of p in its current output format,
// in a fixed order (the Rust test calls them in the same order).
func nhContentValues(ctx context.Context, h *HugoSites, idx int, p *pageState, norm func(string) string) map[string]any {
	m := map[string]any{"page": idx}
	if p.pco != nil {
		m["pcoFormat"] = p.pco.po.f.Name
	}
	m["toc"] = string(p.TableOfContents(ctx))
	c, err := p.Content(ctx)
	if err != nil {
		m["contentErr"] = nhErr(err, norm)
	} else {
		m["content"] = string(c.(template.HTML))
	}
	cws, err := p.ContentWithoutSummary(ctx)
	m["contentWithoutSummary"] = string(cws)
	m["contentWithoutSummaryErr"] = nhErr(err, norm)
	r, err := p.Markup().Render(ctx)
	if err != nil {
		m["summaryErr"] = nhErr(err, norm)
	} else {
		s, err := r.Summary(ctx)
		m["summaryErr"] = nhErr(err, norm)
		m["summaryType"] = s.Type
	}
	m["summary"] = string(p.Summary(ctx))
	m["truncated"] = p.Truncated(ctx)
	m["plain"] = p.Plain(ctx)
	words := p.PlainWords(ctx)
	if words == nil {
		words = []string{}
	}
	m["plainWords"] = words
	m["wordCount"] = p.WordCount(ctx)
	m["fuzzyWordCount"] = p.FuzzyWordCount(ctx)
	m["readingTime"] = p.ReadingTime(ctx)
	m["len"] = p.Len(ctx)
	if fr := p.Fragments(ctx); fr != nil {
		ids := append([]string{}, fr.Identifiers...)
		m["fragments"] = ids
		m["fragmentsHTML13"] = func() string { s, _ := fr.ToHTML(1, 3, true); return string(s) }()
	}
	m["variations"] = p.pageOutputTemplateVariationsState.Load()
	m["fatal"] = nhErr(h.fatalErrorHandler.getErr(), norm)
	return m
}

var _ page.Summary

// nhFNV is FNV-1a 64 as 16 hex digits (hsupport.FNV).
func nhFNV(s string) string {
	var h uint64 = 0xcbf29ce484222325
	for i := 0; i < len(s); i++ {
		h ^= uint64(s[i])
		h *= 0x100000001b3
	}
	return fmt.Sprintf("%016x", h)
}

// nhHashValues replaces the strings of a values dump by their FNV hash and
// length (large sites), and the plain words by the hash of the words joined
// with NUL bytes.
func nhHashValues(m map[string]any) {
	for _, k := range []string{"toc", "content", "contentWithoutSummary", "summary", "plain", "fragmentsHTML13"} {
		if s, ok := m[k].(string); ok {
			m[k] = map[string]any{"fnv": nhFNV(s), "len": len(s)}
		}
	}
	if w, ok := m["plainWords"].([]string); ok {
		var b []byte
		for _, s := range w {
			b = append(b, s...)
			b = append(b, 0)
		}
		m["plainWords"] = map[string]any{"fnv": nhFNV(string(b)), "len": len(w)}
	}
}

// NHOracleContent runs process and assemble, then, like the render loop,
// for every site and render format shifts all pages (preparePagesForRender)
// and reads every content value of every page (with a file) of the rendering
// site. Hook and shortcode template executions are recorded.
func NHOracleContent(h *HugoSites, norm func(string) string, hash bool) (map[string]any, error) {
	ctx := context.Background()
	infol := h.Log.InfoCommand("build")
	conf := &BuildCfg{WhatChanged: &WhatChanged{needsPagesAssembly: true}}
	h.buildCounters = &buildCounters{}
	for _, s := range h.Sites {
		s.state = siteStateInit
	}
	if err := h.process(ctx, infol, conf, func(conf *BuildCfg) error { return h.initSites(conf) }); err != nil {
		return nil, fmt.Errorf("process: %w", err)
	}
	if err := h.assemble(ctx, infol, conf); err != nil {
		return nil, fmt.Errorf("assemble: %w", err)
	}
	for _, s := range h.Sites {
		s.state = siteStateReady
	}

	pages, index, err := nhPages(h)
	if err != nil {
		return nil, err
	}
	pagesDump := []any{}
	for _, p := range pages {
		pagesDump = append(pagesDump, nhPageDump(p, norm))
	}
	global, sitesDump := nhFormats(h)

	NHRecStart()
	defer NHRecStop()
	steps := []any{}
	i := 0
	for si, s := range h.Sites {
		h.currentSite = s
		for _, f := range s.renderFormats {
			for _, s2 := range h.Sites {
				if err := s2.preparePagesForRender(s == s2, i); err != nil {
					return nil, err
				}
			}
			vals := []any{}
			for pi, p := range pages {
				if p.s != s {
					continue
				}
				v := nhContentValues(ctx, h, pi, p, norm)
				if hash {
					nhHashValues(v)
				}
				vals = append(vals, v)
			}
			steps = append(steps, map[string]any{"site": si, "format": f.Name, "idx": i, "values": vals})
			i++
		}
	}
	NHRecStop()

	return map[string]any{
		"pages":         pagesDump,
		"renderFormats": global,
		"sites":         sitesDump,
		"steps":         steps,
		"records":       nhRecordsDump(index),
	}, nil
}

// NHOracleHookrecBuild runs a real build (render included, into the site's
// public dir) with the recorder on, and dumps the records plus the rendered
// content caches of every page (with a file) after the build: for each page
// and cache entry (the output format of the content output that rendered it),
// content, summary and plain values.
func NHOracleHookrecBuild(h *HugoSites, norm func(string) string) (map[string]any, error) {
	NHRecStart()
	err := h.Build(BuildCfg{NoBuildLock: true})
	NHRecStop()
	if err != nil {
		return nil, err
	}
	pages, index, err := nhPages(h)
	if err != nil {
		return nil, err
	}
	pagesDump := []any{}
	for _, p := range pages {
		pagesDump = append(pagesDump, nhPageDump(p, norm))
	}
	global, sitesDump := nhFormats(h)

	// The rendered content caches, by page and output format name.
	caches := []any{}
	for pi, p := range pages {
		pm := p.s.pageMap
		prefix := p.m.content.pi.sourceKey + "/"
		var formats []string
		for _, f := range h.renderFormats {
			formats = append(formats, f.Name)
		}
		sort.Strings(formats)
		seen := map[string]bool{}
		for _, fn := range formats {
			if seen[fn] {
				continue
			}
			seen[fn] = true
			key := prefix + fn
			m := map[string]any{"page": pi, "format": fn}
			if v, ok := pm.cacheContentRendered.Get(context.Background(), key); ok {
				m["content"] = string(v.Value.content)
				m["contentWithoutSummary"] = string(v.Value.contentWithoutSummary)
				m["summary"] = string(v.Value.summary.Text)
				m["summaryType"] = v.Value.summary.Type
				m["truncated"] = v.Value.summary.Truncated
			}
			if v, ok := pm.contentTableOfContents.Get(context.Background(), key); ok {
				m["toc"] = string(v.Value.tableOfContentsHTML)
			}
			if v, ok := pm.cacheContentPlain.Get(context.Background(), key); ok {
				m["plain"] = v.Value.plain
				m["wordCount"] = v.Value.wordCount
				m["readingTime"] = v.Value.readingTime
			}
			if len(m) > 2 {
				caches = append(caches, m)
			}
		}
	}

	return map[string]any{
		"pages":         pagesDump,
		"renderFormats": global,
		"sites":         sitesDump,
		"caches":        caches,
		"records":       nhRecordsDump(index),
	}, nil
}
