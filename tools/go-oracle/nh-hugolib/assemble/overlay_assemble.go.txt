package hugolib

// Added to package hugolib by the nh-hugolib assemble oracle (go run -overlay);
// never part of a real build.

import (
	"context"
	"fmt"
	"path"
	"sort"
	"strings"

	"github.com/neohugo/neohugo/hugolib/doctree"
	"github.com/neohugo/neohugo/resources/kinds"
	"github.com/neohugo/neohugo/resources/page"
	"github.com/neohugo/neohugo/resources/resource"
)

// NHOracleAssemble runs the capture and the assembly phases of a build
// (process, assemble) and dumps the result: every page's meta after
// setMetaPost, the page set of each site in walk order, the page outputs
// and target paths, the collections (queried in a fixed order), the
// taxonomies, .Data, GetPage and GetTerms results, the resources and the
// query cache keys. norm normalises file names; enc encodes a Go value as
// typed JSON.
func NHOracleAssemble(h *HugoSites, norm func(string) string, enc func(any) any) (map[string]any, error) {
	conf := &BuildCfg{WhatChanged: &WhatChanged{needsPagesAssembly: true}}
	l := h.Log.InfoCommand("build")
	if err := h.process(context.Background(), l, conf, func(*BuildCfg) error { return nil }); err != nil {
		return nil, err
	}
	// The main sections applyAggregates picks from when they are not configured: a map range
	// over the page count per section (before removeShouldNotBuild; pages are counted in the
	// capture's page tree), so any section with the highest count. Record ties as a set.
	var mainSectionsTie [][]string
	for _, s := range h.Sites {
		if s.conf.C.IsMainSectionsSet() {
			mainSectionsTie = append(mainSectionsTie, nil)
			continue
		}
		counts := map[string]int{}
		w := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: s.pageMap.treePages,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				if p := n.(*pageState); p.IsPage() {
					counts[p.Section()]++
				}
				return false, nil
			},
		}
		if err := w.Walk(context.Background()); err != nil {
			return nil, err
		}
		max := 0
		for k, c := range counts {
			if k != "" && c > max {
				max = c
			}
		}
		var cands []string
		for k, c := range counts {
			if k != "" && c == max {
				cands = append(cands, k)
			}
		}
		sort.Strings(cands)
		if len(cands) < 2 {
			cands = nil
		}
		mainSectionsTie = append(mainSectionsTie, cands)
	}

	if err := h.assemble(context.Background(), l, conf); err != nil {
		return nil, err
	}
	for _, s := range h.Sites {
		s.state = siteStateReady
	}

	var pages []*pageState
	seen := map[*pageState]int{}
	idx := func(p *pageState) int {
		if i, ok := seen[p]; ok {
			return i
		}
		seen[p] = len(pages)
		pages = append(pages, p)
		return len(pages) - 1
	}
	anyIdx := func(p page.Page) any {
		if p == nil {
			return nil
		}
		switch v := p.(type) {
		case *pageState:
			return idx(v)
		case pageWithWeight0:
			return idx(v.pageState)
		case *pageWithOrdinal:
			return idx(v.pageState)
		case pageWithOrdinal:
			return idx(v.pageState)
		}
		panic(fmt.Sprintf("unknown page type %T", p))
	}
	listDump := func(ps page.Pages) any {
		out := []any{}
		for _, p := range ps {
			e := map[string]any{"p": anyIdx(p)}
			switch v := p.(type) {
			case pageWithWeight0:
				e["w0"] = v.weight0
			case *pageWithOrdinal:
				e["ord"] = v.ordinal
			case pageWithOrdinal:
				e["ord"] = v.ordinal
			}
			out = append(out, e)
		}
		return out
	}
	weightedDump := func(wp page.WeightedPages) any {
		if wp == nil {
			return nil
		}
		out := []any{}
		for _, w := range wp {
			out = append(out, map[string]any{"w": w.Weight, "p": anyIdx(w.Page)})
		}
		return out
	}
	taxonomyDump := func(t page.Taxonomy) any {
		if t == nil {
			return nil
		}
		keys := make([]string, 0, len(t))
		for k := range t {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		out := []any{}
		for _, k := range keys {
			out = append(out, []any{k, weightedDump(t[k])})
		}
		return out
	}
	resultDump := func(p page.Page, err error) any {
		if err != nil {
			return map[string]any{"err": err.Error()}
		}
		return map[string]any{"p": anyIdx(p)}
	}

	var rendersFormats []string
	for _, f := range h.renderFormats {
		rendersFormats = append(rendersFormats, f.Name)
	}

	// Pass 1: the page set of every site (walk order), bundled pages, the
	// collections and the queries, in a fixed order.
	siteDumps := []any{}
	for si, s := range h.Sites {
		m := s.pageMap
		sd := map[string]any{"lang": s.Lang()}
		var rf []string
		for _, f := range s.renderFormats {
			rf = append(rf, f.Name)
		}
		sd["renderFormats"] = rf
		if s.home != nil {
			sd["home"] = idx(s.home)
		} else {
			sd["home"] = nil
		}
		sd["lastmod"] = enc(s.lastmod)
		if tie := mainSectionsTie[si]; tie != nil {
			// Go's random choice among tie; record the set.
			ms := s.conf.C.MainSections
			if len(ms) != 1 || sort.SearchStrings(tie, ms[0]) == len(tie) || tie[sort.SearchStrings(tie, ms[0])] != ms[0] {
				return nil, fmt.Errorf("main sections %v not in %v", ms, tie)
			}
			sd["mainSections"] = nil
			sd["mainSectionsTie"] = tie
		} else {
			sd["mainSections"] = s.conf.C.MainSections
			sd["mainSectionsTie"] = nil
		}

		var walk []any
		var nodes []*pageState
		w := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: m.treePages,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				p := n.(*pageState)
				walk = append(walk, []any{key, idx(p)})
				nodes = append(nodes, p)
				return false, nil
			},
		}
		if err := w.Walk(context.Background()); err != nil {
			return nil, err
		}
		sd["walk"] = walk

		var bundled []any
		rw := &doctree.NodeShiftTreeWalker[contentNodeI]{
			Tree: m.treeResources,
			Handle: func(key string, n contentNodeI, match doctree.DimensionFlag) (bool, error) {
				if rs, ok := n.(*resourceSource); ok {
					if p, ok := rs.r.(*pageState); ok {
						bundled = append(bundled, []any{key, idx(p)})
					}
				}
				return false, nil
			},
		}
		if err := rw.Walk(context.Background()); err != nil {
			return nil, err
		}
		sd["bundled"] = bundled

		// Collections: .Pages and .RegularPages of every node in walk order; for
		// every other term .RegularPages first (they share a cache key).
		colls := []any{}
		for i, p := range nodes {
			var pagesL, regularL page.Pages
			if p.Kind() == kinds.KindTerm && i%2 == 1 {
				regularL = p.RegularPages()
				pagesL = p.Pages()
			} else {
				pagesL = p.Pages()
				regularL = p.RegularPages()
			}
			colls = append(colls, map[string]any{"p": idx(p), "pages": listDump(pagesL), "regularPages": listDump(regularL)})
		}
		sd["collections"] = colls
		sd["sitePages"] = listDump(s.Pages())
		sd["siteRegularPages"] = listDump(s.RegularPages())

		// Site.Taxonomies.
		tl := s.Taxonomies()
		tkeys := make([]string, 0, len(tl))
		for k := range tl {
			tkeys = append(tkeys, k)
		}
		sort.Strings(tkeys)
		taxos := []any{}
		for _, k := range tkeys {
			taxos = append(taxos, []any{k, taxonomyDump(tl[k])})
		}
		sd["taxonomies"] = taxos

		// .Data of every node.
		datas := []any{}
		for _, p := range nodes {
			d := p.Data().(page.Data)
			keys := make([]string, 0, len(d))
			for k := range d {
				keys = append(keys, k)
			}
			sort.Strings(keys)
			dd := []any{}
			for _, k := range keys {
				var v any
				switch x := d[k].(type) {
				case string:
					v = map[string]any{"s": x}
				case page.WeightedPages:
					v = map[string]any{"weighted": weightedDump(x)}
				case page.Taxonomy:
					v = map[string]any{"taxonomy": taxonomyDump(x)}
				case func() page.Pages:
					v = map[string]any{"func": true}
				default:
					v = map[string]any{"other": fmt.Sprintf("%T", x)}
				}
				dd = append(dd, []any{k, v})
			}
			datas = append(datas, map[string]any{"p": idx(p), "data": dd})
		}
		sd["data"] = datas

		// GetTerms of every node for every taxonomy.
		terms := []any{}
		for _, p := range nodes {
			for _, v := range m.cfg.taxonomyConfig.views {
				ts := p.GetTerms(v.plural)
				if len(ts) == 0 {
					continue
				}
				terms = append(terms, map[string]any{"p": idx(p), "taxonomy": v.plural, "terms": listDump(ts)})
			}
		}
		sd["terms"] = terms

		// GetPage.
		refs := map[string]bool{}
		for _, r := range []string{"", "/", "home", "section", "/_index.md", "_index.md", "index.md", "404", "/404", "sitemap", "/nope", "nope.md", "./x", "../x", ".", "..", "/tags/", "tags", "TAGS", "/categories", "p.md"} {
			refs[r] = true
		}
		for _, p := range nodes {
			refs[p.Path()] = true
			refs[p.Path()+"/"] = true
			refs[p.PathInfo().Path()] = true
			refs[strings.ToUpper(p.Path())] = true
			refs[p.PathInfo().BaseNameNoIdentifier()] = true
			refs[strings.TrimPrefix(p.Path(), "/")] = true
			refs[p.PathInfo().Unnormalized().Path()] = true
			if p.File() != nil {
				refs[p.File().Path()] = true
				refs[p.File().LogicalName()] = true
			}
		}
		refList := make([]string, 0, len(refs))
		for r := range refs {
			refList = append(refList, r)
		}
		sort.Strings(refList)
		getPages := []any{}
		for _, r := range refList {
			pp, err := s.getPageForRefs(r)
			getPages = append(getPages, []any{r, resultDump(pp, err)})
		}
		for _, rs := range [][]string{{"section", "blog"}, {"home", ""}, {"home", "/"}, {"page", "/about"}, {"section", ""}, {"a", "b", "c"}, {"section", "s"}, {"taxonomy", "tags"}} {
			pp, err := s.getPageForRefs(rs...)
			getPages = append(getPages, []any{strings.Join(rs, "|"), resultDump(pp, err)})
		}
		sd["getPage"] = getPages

		// Page-relative lookups (getPage and getPageRef with a context page).
		ctxs := []any{}
		for _, p := range nodes {
			pi := p.PathInfo()
			crefs := []string{"./" + pi.BaseNameNoIdentifier(), "../" + path.Base(pi.ContainerDir()), pi.BaseNameNoIdentifier() + ".md", ".", "..", "./index.md", "../_index.md"}
			for _, r := range crefs {
				pp, err := s.getPage(p, r)
				pr, err2 := s.getPageRef(p, r)
				ctxs = append(ctxs, []any{idx(p), r, resultDump(pp, err), resultDump(pr, err2)})
			}
		}
		sd["getPageCtx"] = ctxs

		// Resources of every node (created in assembleResources).
		ress := []any{}
		for _, p := range nodes {
			rr := m.getOrCreateResourcesForPage(p)
			if len(rr) == 0 {
				continue
			}
			out := []any{}
			for _, r := range rr {
				if rp, ok := r.(*pageState); ok {
					out = append(out, map[string]any{"page": idx(rp)})
					continue
				}
				out = append(out, map[string]any{
					"type":         r.ResourceType(),
					"name":         r.Name(),
					"nameNorm":     resource.NameNormalizedOrName(r),
					"title":        r.Title(),
					"params":       enc(r.Params()),
					"relPermalink": r.RelPermalink(),
					"permalink":    r.Permalink(),
				})
			}
			ress = append(ress, map[string]any{"p": idx(p), "resources": out})
		}
		sd["resources"] = ress

		keys := m.cachePages1.Keys()
		sort.Strings(keys)
		sd["cachePages1"] = keys
		keys = m.cacheGetTerms.Keys()
		sort.Strings(keys)
		sd["cacheGetTerms"] = keys
		keys = m.cacheResources.Keys()
		sort.Strings(keys)
		sd["cacheResources"] = keys

		_ = si
		siteDumps = append(siteDumps, sd)
	}

	cascadeDump := func(pm *pageMeta) any {
		c := pm.pageConfig.CascadeCompiled
		if c == nil {
			return nil
		}
		out := []any{}
		c.Range(func(k page.PageMatcher, v page.PageMatcherParamsConfig) bool {
			out = append(out, map[string]any{
				"target": map[string]any{"path": k.Path, "kind": k.Kind, "lang": k.Lang, "environment": k.Environment},
				"params": enc(v.Params),
				"fields": enc(v.Fields),
			})
			return true
		})
		return out
	}
	ofDump := func(of page.OutputFormat) any {
		return map[string]any{"name": of.Format.Name, "rel": of.Rel, "relPermalink": of.RelPermalink(), "permalink": of.Permalink()}
	}
	strs := func(s []string) []string {
		if s == nil {
			return []string{}
		}
		return s
	}

	// Pass 2: every page reached (the list grows while dumping: tree
	// relations may reach pages outside the walks).
	pageDumps := []any{}
	for i := 0; i < len(pages); i++ {
		p := pages[i]
		m := p.m
		pc := m.pageConfig
		file := ""
		if m.f != nil {
			file = norm(m.f.Filename())
		}
		var cfgFormats []string
		for _, f := range pc.ConfiguredOutputFormats {
			cfgFormats = append(cfgFormats, f.Name)
		}
		pd := map[string]any{
			"site":           p.s.languagei,
			"kind":           p.Kind(),
			"path":           p.Path(),
			"pathInfo":       p.PathInfo().Path(),
			"lang":           p.Lang(),
			"file":           file,
			"title":          p.Title(),
			"linkTitle":      p.LinkTitle(),
			"name":           p.Name(),
			"type":           p.Type(),
			"section":        p.Section(),
			"layout":         p.Layout(),
			"description":    p.Description(),
			"summary":        pc.Summary,
			"slug":           pc.Slug,
			"url":            pc.URL,
			"weight":         pc.Weight,
			"draft":          pc.Draft,
			"isCJK":          pc.IsCJKLanguage,
			"translationKey": pc.TranslationKey,
			"keywords":       strs(pc.Keywords),
			"aliases":        strs(pc.Aliases),
			"outputs":        strs(pc.Outputs),
			"outputFormats":  strs(cfgFormats),
			"markup":         pc.Content.Markup,
			"mediaType":      pc.ContentMediaType.Type,
			"term":           m.term,
			"singular":       m.singular,
			"resourcePath":   m.resourcePath,
			"bundled":        m.bundled,
			"standalone":     m.standaloneOutputFormat.Name,
			"dates": map[string]any{
				"date":        enc(pc.Dates.Date),
				"lastmod":     enc(pc.Dates.Lastmod),
				"publishDate": enc(pc.Dates.PublishDate),
				"expiryDate":  enc(pc.Dates.ExpiryDate),
			},
			"build":         map[string]any{"list": pc.Build.List, "render": pc.Build.Render, "publishResources": pc.Build.PublishResources},
			"sitemap":       map[string]any{"changeFreq": pc.Sitemap.ChangeFreq, "priority": enc(pc.Sitemap.Priority), "filename": pc.Sitemap.Filename, "disable": pc.Sitemap.Disable},
			"params":        enc(pc.Params),
			"cascade":       cascadeDump(m),
			"resourcesMeta": len(pc.ResourcesMeta),
			"noLink":        m.noLink(),
			"noRender":      m.noRender(),
			"shouldList":    []bool{m.shouldList(false), m.shouldList(true)},
			"setMetaPost":   m.setMetaPostCount,
		}

		if err := p.initPage(); err != nil {
			pd["initErr"] = err.Error()
			pageDumps = append(pageDumps, pd)
			continue
		}

		d := p.targetPathDescriptor
		if d.Path == nil {
			// The zero pagePaths of a page without output formats.
			pd["tpd"] = nil
		} else {
			sec := ""
			if d.Section != nil {
				sec = d.Section.Path()
			}
			pd["tpd"] = map[string]any{
				"kind": d.Kind, "path": d.Path.Path(), "section": sec, "baseName": d.BaseName,
				"prefixFilePath": d.PrefixFilePath, "prefixLink": d.PrefixLink, "forcePrefix": d.ForcePrefix,
				"url": d.URL, "expandedPermalink": d.ExpandedPermalink, "uglyURLs": d.UglyURLs,
			}
		}

		ofs := []any{}
		for _, of := range p.OutputFormats() {
			ofs = append(ofs, ofDump(of))
		}
		pd["pageOutputFormats"] = ofs

		outs := []any{}
		for oi, po := range p.pageOutputs {
			slot := oi
			for j := 0; j < oi; j++ {
				if p.pageOutputs[j] == po {
					slot = j
					break
				}
			}
			tp := po.targetPaths()
			outs = append(outs, map[string]any{
				"slot":      slot,
				"name":      po.f.Name,
				"render":    po.render,
				"paginator": po.paginator != nil,
				"relURL":    po.getRelURL(),
				"target": map[string]any{
					"filename":  tp.TargetFilename,
					"subTarget": tp.SubResourceBaseTarget,
					"subLink":   tp.SubResourceBaseLink,
					"link":      tp.Link,
				},
				"relPermalink": po.RelPermalink(),
				"permalink":    po.Permalink(),
			})
		}
		pd["outputs"] = outs
		pd["currentOutput"] = p.pageOutputIdx

		// Tree navigation.
		if pp := p.Parent(); pp != nil {
			pd["parent"] = anyIdx(pp)
		} else {
			pd["parent"] = nil
		}
		pd["currentSection"] = anyIdx(p.CurrentSection())
		if fs := p.FirstSection(); fs != nil {
			pd["firstSection"] = anyIdx(fs)
		} else {
			pd["firstSection"] = nil
		}
		pd["sectionsPath"] = p.SectionsPath()
		pd["sectionsEntries"] = strs(p.SectionsEntries())
		if p.IsNode() {
			pd["sections"] = listDump(p.Sections())
		}

		pageDumps = append(pageDumps, pd)
	}

	return map[string]any{
		"renderFormats": rendersFormats,
		"sites":         siteDumps,
		"pages":         pageDumps,
	}, nil
}
