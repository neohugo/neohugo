// OracleConsidered is one call of a query's Consider func.
type OracleConsidered struct {
	Ti     *TemplInfo
	Result bool
}

// OracleConsiderLog records the Consider calls of one lookup.
type OracleConsiderLog struct {
	mu          sync.Mutex
	HadConsider bool
	Calls       []OracleConsidered
}

// OracleHookSet holds the lookup recorders.
type OracleHookSet struct {
	Pages     func(s *TemplateStore, q TemplateQuery, l *OracleConsiderLog, res *TemplInfo)
	Shortcode func(s *TemplateStore, q TemplateQuery, l *OracleConsiderLog, res *TemplInfo, err error)
	Partial   func(s *TemplateStore, name string, res *TemplInfo)
	ByName    func(s *TemplateStore, name string, res *TemplInfo)
}

// OracleHooks is set by the oracle while recording.
var OracleHooks *OracleHookSet

// OracleWrapConsider wraps q.Consider so that every call is logged.
func OracleWrapConsider(q TemplateQuery) (TemplateQuery, *OracleConsiderLog) {
	l := &OracleConsiderLog{HadConsider: q.Consider != nil}
	f := q.Consider
	q.Consider = func(c *TemplInfo) bool {
		r := true
		if f != nil {
			r = f(c)
		}
		l.mu.Lock()
		l.Calls = append(l.Calls, OracleConsidered{Ti: c, Result: r})
		l.mu.Unlock()
		return r
	}
	return q, l
}

	return ti.category
}

func (ti *TemplInfo) OracleBase() *TemplInfo {

}

// OracleDesc is fmt %+v of a descriptor.
func OracleDesc(d TemplateDescriptor) string {
	return fmt.Sprintf("%+v", d)
}

var oracleIDCache sync.Map // *TemplateStore.treeMain -> map[*TemplInfo]string

func oracleIDs(s *TemplateStore) map[*TemplInfo]string {
	if v, ok := oracleIDCache.Load(s.treeMain); ok {
		return v.(map[*TemplInfo]string)
	}
	ids := map[*TemplInfo]string{}
	_ = s.treeMain.Walk(func(k string, v map[nodeKey]*TemplInfo) (bool, error) {
		for nk, vv := range v {
			ids[vv] = fmt.Sprintf("main|%s|%s|%s|%s", k, nk.c, vv.PathInfo.Path(), OracleDesc(nk.d))
			if vv.baseVariants != nil {
				_ = vv.baseVariants.Walk(func(bk string, bv map[TemplateDescriptor]*TemplWithBaseApplied) (bool, error) {
					for bd, tb := range bv {
						ids[tb.Template] = fmt.Sprintf("variant|%s|%s|%s|base=%s|%s", k, vv.PathInfo.Path(), OracleDesc(vv.D), bk, OracleDesc(bd))
					}
					return false, nil
				})
			}
		}
		return false, nil
	})
	_ = s.treeShortcodes.Walk(func(k string, v map[string]map[TemplateDescriptor]*TemplInfo) (bool, error) {
		for name, m := range v {
			for d, vv := range m {
				ids[vv] = fmt.Sprintf("shortcode|%s|%s|%s|%s", k, name, vv.PathInfo.Path(), OracleDesc(d))
			}
		}
		return false, nil
	})
	oracleIDCache.Store(s.treeMain, ids)
	return ids
}

// OracleTemplID identifies a template of the store ("" for nil).
func OracleTemplID(s *TemplateStore, ti *TemplInfo) string {
	if ti == nil {
		return ""
	}
	if id, ok := oracleIDs(s)[ti]; ok {
		return id
	}
	p := ""
	if ti.PathInfo != nil {
		p = ti.PathInfo.Path()
	}
	return fmt.Sprintf("other|%s|%s", p, OracleDesc(ti.D))
}

// OracleCalls returns the logged calls with template ids.
func (l *OracleConsiderLog) OracleCalls(s *TemplateStore) [][]any {
	l.mu.Lock()
	defer l.mu.Unlock()
	out := [][]any{}
	for _, c := range l.Calls {
		out = append(out, []any{OracleTemplID(s, c.Ti), c.Result})
	}
	return out
}

func oracleParseInfo(pi ParseInfo) map[string]any {
	return map[string]any{
		"isInner":   pi.IsInner,
		"hasReturn": pi.HasReturn,
		"version":   pi.Config.Version,
	}
}

func oracleFilename(ti *TemplInfo) string {
	if ti.Fi == nil {
		return ""
	}
	return ti.Fi.Meta().Filename
}

// oracleNameCounterRe matches the counter doParseTemplate appends to a
// template name that is taken ("_default/taxonomy.html-2"); which template
// gets which number depends on Go's map order, so dumps show "-N".
var oracleNameCounterRe = regexp.MustCompile(`^(.*\.[A-Za-z0-9]+)-[0-9]+($|\$.*)`)

// OracleNormName replaces a name counter with "-N".
func OracleNormName(name string) string {
	return oracleNameCounterRe.ReplaceAllString(name, "${1}-N${2}")
}

func oracleName(ti *TemplInfo) string {
	if ti.Template == nil && ti.PathInfo == nil {
		return "<nil>"
	}
	return OracleNormName(ti.Name())
}

func oracleEntry(s *TemplateStore, ti *TemplInfo) map[string]any {
	p := ""
	if ti.PathInfo != nil {
		p = ti.PathInfo.Path()
	}
	isText := false
	if ti.Template != nil {
		isText = isText2(ti.Template)
	}
	return map[string]any{
		"id":           OracleTemplID(s, ti),
		"category":     ti.category.String(),
		"subCategory":  ti.subCategory.String(),
		"name":         oracleName(ti),
		"path":         p,
		"desc":         OracleDesc(ti.D),
		"noBaseOf":     ti.noBaseOf,
		"legacyMapped": ti.isLegacyMapped,
		"filename":     oracleFilename(ti),
		"parseInfo":    oracleParseInfo(ti.ParseInfo),
		"hasTemplate":  ti.Template != nil,
		"isText":       isText,
		"contentHash":  hashing.XxHashFromStringHexEncoded(ti.content),
	}
}

func isText2(t interface{ Name() string }) bool {
	_, ok := t.(*texttemplate.Template)
	return ok
}

func oracleSortEntries(es []map[string]any) {
	sort.Slice(es, func(i, j int) bool { return es[i]["id"].(string) < es[j]["id"].(string) })
}

// OracleDump dumps the store: every template of the main tree (with its base
// variants), the shortcodes tree, templatesByPath and shortcodesByName, the
// function names and the layouts walk.
func (s *TemplateStore) OracleDump() (map[string]any, error) {
	var main []map[string]any
	_ = s.treeMain.Walk(func(k string, v map[nodeKey]*TemplInfo) (bool, error) {
		var es []map[string]any
		for _, vv := range v {
			e := oracleEntry(s, vv)
			e["key"] = k
			var variants []map[string]any
			if vv.baseVariants != nil {
				_ = vv.baseVariants.Walk(func(bk string, bv map[TemplateDescriptor]*TemplWithBaseApplied) (bool, error) {
					for bd, tb := range bv {
						ve := oracleEntry(s, tb.Template)
						ve["baseKey"] = bk
						ve["baseDesc"] = OracleDesc(bd)
						ve["baseId"] = OracleTemplID(s, tb.Base)
						ve["overlayId"] = OracleTemplID(s, tb.Overlay)
						variants = append(variants, ve)
					}
					return false, nil
				})
			}
			oracleSortEntries(variants)
			e["variants"] = variants
			var overlays []string
			for _, o := range vv.overlays {
				overlays = append(overlays, OracleTemplID(s, o))
			}
			sort.Strings(overlays)
			e["overlays"] = overlays
			es = append(es, e)
		}
		oracleSortEntries(es)
		main = append(main, es...)
		return false, nil
	})
	var shortcodes []map[string]any
	_ = s.treeShortcodes.Walk(func(k string, v map[string]map[TemplateDescriptor]*TemplInfo) (bool, error) {
		var es []map[string]any
		for name, m := range v {
			for _, vv := range m {
				e := oracleEntry(s, vv)
				e["key"] = k
				e["scName"] = name
				es = append(es, e)
			}
		}
		oracleSortEntries(es)
		shortcodes = append(shortcodes, es...)
		return false, nil
	})
	byPath := [][]any{}
	s.templatesByPath.ForEeach(func(k string, v *TemplInfo) bool {
		id := OracleTemplID(s, v)
		if strings.HasPrefix(k, "__hdeferred/") {
			// Which template owns a deferred template (the first transformed
			// one with the same deferred body) depends on Go's map order.
			id = "deferred"
		}
		byPath = append(byPath, []any{k, id, oracleName(v)})
		return true
	})
	sort.Slice(byPath, func(i, j int) bool { return byPath[i][0].(string) < byPath[j][0].(string) })
	byName := [][]any{}
	s.shortcodesByName.ForEeach(func(k string, v *TemplInfo) bool {
		byName = append(byName, []any{k, OracleTemplID(s, v)})
		return true
	})
	sort.Slice(byName, func(i, j int) bool { return byName[i][0].(string) < byName[j][0].(string) })

	var funcNames []string
	for k := range s.siteOpts.TemplateFuncs {
		funcNames = append(funcNames, k)
	}
	sort.Strings(funcNames)

	var walk [][]any
	err := helpers.Walk(s.opts.Fs, "", func(pth string, fi hugofs.FileMetaInfo) error {
		m := fi.Meta()
		pi := ""
		if m.PathInfo != nil {
			pi = m.PathInfo.Path()
		}
		walk = append(walk, []any{pth, fi.IsDir(), m.Filename, m.ModuleOrdinal, pi})
		return nil
	})
	if err != nil {
		return nil, err
	}

	// Both depend on Go's map order: sorted.
	var templates []string
	for t := range s.templates() {
		templates = append(templates, OracleTemplID(s, t))
	}
	sort.Strings(templates)

	var unused []string
	for _, t := range s.UnusedTemplates() {
		unused = append(unused, OracleTemplID(s, t))
	}
	sort.Strings(unused)

	return map[string]any{
		"main":             main,
		"shortcodes":       shortcodes,
		"byPath":           byPath,
		"shortcodesByName": byName,
		"funcNames":        funcNames,
		"layoutsWalk":      walk,
		"templates":        templates,
		"unused":           unused,
	}, nil
}

func oracleDumpNamespace(label string, tt *texttemplate.Template, main map[string]*parse.Tree) map[string]any {
	var list [][]string
	shared := 0
	for _, t := range tt.Templates() {
		if main != nil {
			if mt, ok := main[t.Name()]; ok && mt == t.Tree {
				shared++
				continue
			}
		}
		s := "<nil>"
		if t.Tree != nil && t.Root != nil {
			s = t.Root.String()
		}
		list = append(list, []string{OracleNormName(t.Name()), s})
	}
	sort.Slice(list, func(i, j int) bool {
		if list[i][0] != list[j][0] {
			return list[i][0] < list[j][0]
		}
		return list[i][1] < list[j][1]
	})
	return map[string]any{
		"label":     label,
		"shared":    shared,
		"templates": list,
	}
}

func oracleTrees(tt *texttemplate.Template) map[string]*parse.Tree {
	m := map[string]*parse.Tree{}
	for _, t := range tt.Templates() {
		m[t.Name()] = t.Tree
	}
	return m
}

// OracleNamespaces dumps every tree of the store's namespaces: the main html
// and text namespaces, and each base-applied variant's namespace (only its
// trees that are not the main namespace's tree of the same name).
func (s *TemplateStore) OracleNamespaces() []any {
	htmlText := s.tns.parseHTML.OracleText()
	mainHTML := oracleTrees(htmlText)
	mainText := oracleTrees(s.tns.parseText)
	out := []any{
		oracleDumpNamespace("html", htmlText, nil),
		oracleDumpNamespace("text", s.tns.parseText, nil),
	}
	var variants []map[string]any
	for t := range s.templates() {
		if t.base == nil {
			continue
		}
		label := OracleTemplID(s, t)
		switch tt := t.Template.(type) {
		case *htmltemplate.Template:
			variants = append(variants, oracleDumpNamespace(label, tt.OracleText(), mainHTML))
		case *texttemplate.Template:
			variants = append(variants, oracleDumpNamespace(label, tt, mainText))
		}
	}
	sort.Slice(variants, func(i, j int) bool {
		return variants[i]["label"].(string) < variants[j]["label"].(string)
	})
	for _, v := range variants {
		out = append(out, v)
	}
	return out
}

// OracleReverseOrder reverses the order oracleNodeKeys and oracleDescKeys
// give.
var OracleReverseOrder bool

func oracleDescLess(a, b TemplateDescriptor) bool {
	sa := []string{a.Kind, a.LayoutFromTemplate, a.LayoutFromUser, a.OutputFormat, a.MediaType, a.Lang, a.Variant1, a.Variant2}
	sb := []string{b.Kind, b.LayoutFromTemplate, b.LayoutFromUser, b.OutputFormat, b.MediaType, b.Lang, b.Variant1, b.Variant2}
	for i := range sa {
		if sa[i] != sb[i] {
			return sa[i] < sb[i]
		}
	}
	ba := []bool{a.LayoutFromUserMustMatch, a.IsPlainText, a.AlwaysAllowPlainText}
	bb := []bool{b.LayoutFromUserMustMatch, b.IsPlainText, b.AlwaysAllowPlainText}
	for i := range ba {
		if ba[i] != bb[i] {
			return !ba[i]
		}
	}
	return false
}

// oracleDescKeys returns the keys in the order of the Rust store's
// BTreeMap<TemplateDescriptor, _> (derived Ord: fields in declaration order,
// false before true).
func oracleDescKeys[V any](m map[TemplateDescriptor]V) []TemplateDescriptor {
	keys := make([]TemplateDescriptor, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Slice(keys, func(i, j int) bool {
		if OracleReverseOrder {
			return oracleDescLess(keys[j], keys[i])
		}
		return oracleDescLess(keys[i], keys[j])
	})
	return keys
}

// oracleNodeKeys returns the keys in the order of the Rust store's
// BTreeMap<NodeKey, _> (category, then descriptor).
func oracleNodeKeys(m map[nodeKey]*TemplInfo) []nodeKey {
	keys := make([]nodeKey, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	less := func(a, b nodeKey) bool {
		if a.c != b.c {
			return a.c < b.c
		}
		return oracleDescLess(a.d, b.d)
	}
	sort.Slice(keys, func(i, j int) bool {
		if OracleReverseOrder {
			return less(keys[j], keys[i])
		}
		return less(keys[i], keys[j])
	})
	return keys
}
