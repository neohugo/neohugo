package hugolib

// Added to hugolib by the nh-tplimpl oracles (go run -overlay).

import (
	"reflect"
	"sort"

	"github.com/neohugo/neohugo/tpl/tplimpl"
)

// OracleSkipWrite makes renderAndWritePage a no-op (the template is still
// resolved, nothing is executed).
var OracleSkipWrite bool

func oracleMediaTypeDump(m any) map[string]any {
	v := reflect.ValueOf(m)
	suffixes := v.MethodByName("Suffixes").Call(nil)[0].Interface().([]string)
	return map[string]any{
		"type":        v.FieldByName("Type").String(),
		"mainType":    v.FieldByName("MainType").String(),
		"subType":     v.FieldByName("SubType").String(),
		"delimiter":   v.FieldByName("Delimiter").String(),
		"suffix":      v.FieldByName("FirstSuffix").FieldByName("Suffix").String(),
		"fullSuffix":  v.FieldByName("FirstSuffix").FieldByName("FullSuffix").String(),
		"mimeSuffix":  v.FieldByName("mimeSuffix").String(),
		"suffixesCSV": v.FieldByName("SuffixesCSV").String(),
		"suffixes":    suffixes,
	}
}

// OracleSiteConfig dumps the configuration the template store and the
// layouts filesystem depend on.
func OracleSiteConfig(h *HugoSites) map[string]any {
	s := h.Sites[0]
	conf := s.conf
	var formats []any
	for _, f := range conf.OutputFormats.Config {
		formats = append(formats, map[string]any{
			"name":           f.Name,
			"mediaType":      oracleMediaTypeDump(f.MediaType),
			"path":           f.Path,
			"baseName":       f.BaseName,
			"rel":            f.Rel,
			"protocol":       f.Protocol,
			"isPlainText":    f.IsPlainText,
			"isHTML":         f.IsHTML,
			"noUgly":         f.NoUgly,
			"ugly":           f.Ugly,
			"notAlternative": f.NotAlternative,
			"root":           f.Root,
			"permalinkable":  f.Permalinkable,
			"weight":         f.Weight,
		})
	}
	var mediaTypes []any
	suffixSet := map[string]bool{}
	for _, m := range conf.MediaTypes.Config {
		mediaTypes = append(mediaTypes, oracleMediaTypeDump(m))
		for _, sfx := range m.Suffixes() {
			suffixSet[sfx] = true
		}
	}
	var contentSuffixes []string
	for sfx := range suffixSet {
		if conf.ContentTypes.Config.IsContentSuffix(sfx) {
			contentSuffixes = append(contentSuffixes, sfx)
		}
	}
	sort.Strings(contentSuffixes)
	var disabled []string
	for l, b := range conf.C.DisabledLanguages {
		if b {
			disabled = append(disabled, l)
		}
	}
	sort.Strings(disabled)
	pp := s.Conf.PathParser()
	return map[string]any{
		"outputFormats":          formats,
		"mediaTypes":             mediaTypes,
		"contentSuffixes":        contentSuffixes,
		"disabledLanguages":      disabled,
		"languageIndex":          pp.LanguageIndex,
		"defaultContentLanguage": s.Conf.DefaultContentLanguage(),
		"defaultOutputFormat":    conf.DefaultOutputFormat,
		"taxonomies":             conf.Taxonomies,
		"renderHookImage":        conf.Markup.Goldmark.RenderHooks.Image.UseEmbedded,
		"renderHookLink":         conf.Markup.Goldmark.RenderHooks.Link.UseEmbedded,
	}
}

// OracleNewStore creates a template store like NewHugoSites does, with the
// given template funcs.
func OracleNewStore(h *HugoSites, funcs map[string]any) (*tplimpl.TemplateStore, error) {
	s := h.Sites[0]
	return tplimpl.NewStore(
		tplimpl.StoreOptions{
			Fs:                     s.BaseFs.Layouts.Fs,
			Log:                    s.Log,
			DefaultContentLanguage: s.Conf.DefaultContentLanguage(),
			Watching:               s.Conf.Watching(),
			PathParser:             s.Conf.PathParser(),
			OutputFormats:          s.conf.OutputFormats.Config,
			MediaTypes:             s.conf.MediaTypes.Config,
			DefaultOutputFormat:    s.conf.DefaultOutputFormat,
			TaxonomySingularPlural: s.conf.Taxonomies,
			RenderHooks:            s.conf.Markup.Goldmark.RenderHooks,
		}, tplimpl.SiteOptions{
			Site:          s,
			TemplateFuncs: funcs,
		})
}
