package tsupport

// Added to tsupport by the nh-tplimpl oracles (go run -overlay): wires the
// overlay-added functions of tplimpl and hugolib into the API.

import (
	"github.com/neohugo/neohugo/hugolib"
	"github.com/neohugo/neohugo/tpl/tplimpl"
)

func init() {
	Oracle = &API{
		StoreDump: func(s *tplimpl.TemplateStore) (map[string]any, error) {
			return s.OracleDump()
		},
		Namespaces: func(s *tplimpl.TemplateStore) []any {
			return s.OracleNamespaces()
		},
		SiteConfig: hugolib.OracleSiteConfig,
		SetSkipWrite: func(b bool) {
			hugolib.OracleSkipWrite = b
		},
		NewStore: hugolib.OracleNewStore,
		TemplID:  tplimpl.OracleTemplID,
		SetReverseOrder: func(b bool) {
			tplimpl.OracleReverseOrder = b
		},
		InstallLookupHooks: func(r *LookupRecorder) {
			if r == nil {
				tplimpl.OracleHooks = nil
				return
			}
			tplimpl.OracleHooks = &tplimpl.OracleHookSet{
				Pages: func(s *tplimpl.TemplateStore, q tplimpl.TemplateQuery, l *tplimpl.OracleConsiderLog, res *tplimpl.TemplInfo) {
					r.Add(map[string]any{
						"op":          "pages",
						"query":       QueryDump(q),
						"hadConsider": l.HadConsider,
						"considered":  considered(s, l),
						"result":      tplimpl.OracleTemplID(s, res),
					})
				},
				Shortcode: func(s *tplimpl.TemplateStore, q tplimpl.TemplateQuery, l *tplimpl.OracleConsiderLog, res *tplimpl.TemplInfo, err error) {
					e := ""
					if err != nil {
						e = err.Error()
					}
					r.Add(map[string]any{
						"op":          "shortcode",
						"query":       QueryDump(q),
						"hadConsider": l.HadConsider,
						"considered":  considered(s, l),
						"result":      tplimpl.OracleTemplID(s, res),
						"err":         e,
					})
				},
				Partial: func(s *tplimpl.TemplateStore, name string, res *tplimpl.TemplInfo) {
					r.Add(map[string]any{
						"op":     "partial",
						"name":   name,
						"result": tplimpl.OracleTemplID(s, res),
					})
				},
				ByName: func(s *tplimpl.TemplateStore, name string, res *tplimpl.TemplInfo) {
					r.Add(map[string]any{
						"op":     "byName",
						"name":   name,
						"result": tplimpl.OracleTemplID(s, res),
					})
				},
			}
		},
	}
}

// considered is the Consider log of a query that had a Consider func (nil
// otherwise: the store then considers every candidate).
func considered(s *tplimpl.TemplateStore, l *tplimpl.OracleConsiderLog) [][]any {
	if !l.HadConsider {
		return nil
	}
	return l.OracleCalls(s)
}
