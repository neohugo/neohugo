package tplimpl

// Added to package tplimpl by the structure oracle (tools/go-oracle/structure,
// built with `go build -overlay`); never part of a neohugo build.
//
// Template names are normalised to Hugo v0.146 layout names: the store's own
// conversion of legacy paths (fromLegacyPath: `_default/` dropped,
// `partials/` -> `_partials/`, `shortcodes/` -> `_shortcodes/`,
// `<id>-baseof` -> `baseof.<id>`), plus the two mappings the store keeps in
// its descriptors only: the legacy home layout `index` is `home`, and a
// template the store inserted through its legacy taxonomy, term or section
// mapping is named after its tree key and kind (`taxonomy/list.html` ->
// `taxonomy.html`, `term/term.html` -> `term.html`,
// `taxonomy/tag.terms.html` -> `tags/taxonomy.html`).

import (
	"path"
	"strings"
)

type nhStructEntry struct {
	key string
	ti  *TemplInfo
}

// NHStructIndex maps every template of the main tree, and every
// base-applied variant of one, to the tree entry it comes from.
type NHStructIndex struct {
	s *TemplateStore
	m map[*TemplInfo]nhStructEntry
}

// NHStructNewIndex indexes the templates of s.
func NHStructNewIndex(s *TemplateStore) *NHStructIndex {
	idx := &NHStructIndex{s: s, m: map[*TemplInfo]nhStructEntry{}}
	_ = s.treeMain.Walk(func(k string, v map[nodeKey]*TemplInfo) (bool, error) {
		for _, vv := range v {
			idx.m[vv] = nhStructEntry{key: k, ti: vv}
			if vv.baseVariants == nil {
				continue
			}
			_ = vv.baseVariants.Walk(func(_ string, bv map[TemplateDescriptor]*TemplWithBaseApplied) (bool, error) {
				for _, tb := range bv {
					idx.m[tb.Template] = nhStructEntry{key: k, ti: vv}
				}
				return false, nil
			})
		}
		return false, nil
	})
	return idx
}

// NHStructNames are the normalised names of a chosen template and its base
// template, and the files they come from (relative to the layouts
// directory; `_embedded/<name>` for Hugo's embedded templates).
type NHStructNames struct {
	Template, TemplateFile string
	Baseof, BaseofFile     string
}

// Names returns the names of ti, a template LookupPagesLayout returned
// (nil: no template).
func (idx *NHStructIndex) Names(ti *TemplInfo) NHStructNames {
	var n NHStructNames
	if ti == nil {
		return n
	}
	n.Template, n.TemplateFile = idx.name(ti)
	if ti.base != nil {
		n.Baseof, n.BaseofFile = idx.name(ti.base)
	}
	return n
}

func nhFile(ti *TemplInfo, name string) string {
	if ti.subCategory == SubCategoryEmbedded || ti.Fi == nil {
		return "_embedded/" + name
	}
	pi := ti.Fi.Meta().PathInfo
	if u := pi.Unnormalized(); u != nil {
		return u.PathNoLeadingSlash()
	}
	return pi.PathNoLeadingSlash()
}

// name returns the normalised name of a template and its file.
func (idx *NHStructIndex) name(ti *TemplInfo) (string, string) {
	e, ok := idx.m[ti]
	if !ok {
		e = nhStructEntry{key: "", ti: ti}
	}
	t := e.ti
	if t.PathInfo == nil {
		return t.Name(), t.Name()
	}
	var name string
	switch {
	case t.subCategory == SubCategoryEmbedded:
		name = t.PathInfo.PathNoLeadingSlash()
	case t.isLegacyMapped:
		pi := t.PathInfo
		suffix := strings.TrimPrefix(pi.Path(), pi.PathBeforeLangAndOutputFormatAndExt())
		stem := t.D.Kind
		if t.category == CategoryBaseof {
			stem = baseNameBaseof + "." + t.D.Kind
		}
		name = strings.TrimPrefix(path.Join(e.key, stem+suffix), "/")
	default:
		p := t.PathInfo.Path()
		if t.category == CategoryLayout && t.PathInfo.Layout() == "index" {
			dir, file := path.Split(p)
			p = dir + "home" + strings.TrimPrefix(file, "index")
		}
		name = strings.TrimPrefix(p, "/")
	}
	return name, nhFile(t, name)
}

func nhCategory(c Category) string {
	return strings.ToLower(strings.TrimPrefix(c.String(), "Category"))
}

func nhDesc(d TemplateDescriptor) map[string]any {
	m := map[string]any{}
	add := func(k, v string) {
		if v != "" {
			m[k] = v
		}
	}
	add("kind", d.Kind)
	add("layout", d.LayoutFromTemplate)
	add("lang", d.Lang)
	add("outputFormat", d.OutputFormat)
	add("mediaType", d.MediaType)
	add("variant1", d.Variant1)
	add("variant2", d.Variant2)
	if d.IsPlainText {
		m["plainText"] = true
	}
	return m
}

// Layouts lists every template the store read from the layouts directory
// (themes included; embedded and inline templates are not listed), with its
// normalised name, category, tree key and descriptor. A file can be listed
// more than once: the store inserts a legacy file under its v0.146 path and
// again under every legacy mapping that matches it.
func (idx *NHStructIndex) Layouts() []map[string]any {
	out := []map[string]any{}
	add := func(key string, t *TemplInfo) {
		if t.Fi == nil || t.subCategory != SubCategoryMain {
			return
		}
		name, file := idx.name(t)
		e := map[string]any{
			"file":     file,
			"name":     name,
			"category": nhCategory(t.category),
			"key":      key,
			"desc":     nhDesc(t.D),
		}
		if t.isLegacyMapped {
			e["legacy"] = true
		}
		out = append(out, e)
	}
	_ = idx.s.treeMain.Walk(func(k string, v map[nodeKey]*TemplInfo) (bool, error) {
		for _, vv := range v {
			add(k, vv)
		}
		return false, nil
	})
	_ = idx.s.treeShortcodes.Walk(func(k string, v map[string]map[TemplateDescriptor]*TemplInfo) (bool, error) {
		for _, m := range v {
			for _, vv := range m {
				add(k, vv)
			}
		}
		return false, nil
	})
	return out
}
