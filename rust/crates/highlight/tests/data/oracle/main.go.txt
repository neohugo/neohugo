// The Go oracle of neohugo-highlight's fixtures (crate README, "Fixtures"). Build it as a
// module next to a go.mod that requires github.com/neohugo/neohugo and replaces it with this
// checkout; modes: tokens | docs | html (JSON on stdin) | css <style> all|omit | keys | lexers.
package main

import (
	"encoding/json"
	"fmt"
	"os"
	"sort"
	"strings"

	"github.com/alecthomas/chroma/v2"
	"github.com/alecthomas/chroma/v2/formatters/html"
	"github.com/alecthomas/chroma/v2/lexers"
	"github.com/alecthomas/chroma/v2/styles"
	"github.com/neohugo/neohugo/markup/highlight"
	"github.com/neohugo/neohugo/markup/highlight/chromalexers"
)

type item struct {
	Key  string `json:"key"`
	Lang string `json:"lang"`
	Code string `json:"code"`
	Opts any    `json:"opts"`
	// Cfg overrides of the highlighter config (noClasses etc.).
	NoClasses *bool  `json:"noClasses"`
	Style     string `json:"style"`
}

type out struct {
	Key   string  `json:"key"`
	Lexer string  `json:"lexer"`
	Runs  [][]any `json:"runs,omitempty"`
	HTML  string  `json:"html,omitempty"`
	Err   string  `json:"err,omitempty"`
}

func class(t chroma.TokenType) string {
	for t != 0 {
		if cls, ok := chroma.StandardTypes[t]; ok {
			return cls
		}
		t = t.Parent()
	}
	return chroma.StandardTypes[t]
}

func main() {
	switch os.Args[1] {
	case "lexers":
		ls := lexers.GlobalLexerRegistry.Lexers
		sort.Slice(ls, func(i, j int) bool { return ls[i].Config().Name < ls[j].Config().Name })
		for _, l := range ls {
			c := l.Config()
			fmt.Printf("%s\t%s\t%s\n", c.Name, strings.Join(c.Aliases, " "), strings.Join(c.Filenames, " "))
		}
	case "docs":
		var items []struct {
			Key  string `json:"html_key"`
			Lang string `json:"lang"`
			Code string `json:"code"`
			Opts any    `json:"opts"`
		}
		if err := json.NewDecoder(os.Stdin).Decode(&items); err != nil {
			panic(err)
		}
		cfg := highlight.DefaultConfig
		cfg.NoClasses = false
		cfg.Style = "solarized-dark"
		cfg.LineNumbersInTable = false
		cfg.WrapperClass = "highlight not-prose"
		h := highlight.New(cfg)
		type res struct {
			Key  string `json:"key"`
			HTML string `json:"html"`
		}
		var out []res
		for _, it := range items {
			if m, ok := it.Opts.(map[string]any); ok {
				for k, v := range m {
					if arr, ok := v.([]any); ok {
						var r [][2]int
						for _, x := range arr {
							p := x.([]any)
							r = append(r, [2]int{int(p[0].(float64)), int(p[1].(float64))})
						}
						m[k] = r
					}
				}
			}
			s, err := h.Highlight(it.Code, it.Lang, it.Opts)
			if err != nil {
				s = "ERROR: " + err.Error()
			}
			out = append(out, res{it.Key, s})
		}
		enc := json.NewEncoder(os.Stdout)
		enc.SetEscapeHTML(false)
		_ = enc.Encode(out)
	case "css":
		var f *html.Formatter
		if os.Args[3] == "omit" {
			f = html.New(html.WithClasses(true))
		} else {
			f = html.New(html.WithAllClasses(true))
		}
		_ = f.WriteCSS(os.Stdout, styles.Get(os.Args[2]))
	case "keys":
		seen := map[string]bool{}
		type kv struct{ k, kind, v string }
		var rows []kv
		for _, l := range lexers.GlobalLexerRegistry.Lexers {
			c := l.Config()
			for _, n := range append([]string{c.Name}, c.Aliases...) {
				k := strings.ToLower(n)
				if seen["n"+k] {
					continue
				}
				seen["n"+k] = true
				if got := lexers.Get(k); got != nil {
					rows = append(rows, kv{k, "n", got.Config().Name})
				}
			}
			for _, g := range c.Filenames {
				k := g
				if strings.HasPrefix(g, "*.") {
					k = g[2:]
				}
				if strings.ContainsAny(k, "*?[") || seen["f"+k] || seen["n"+strings.ToLower(k)] && lexers.Get(k) == lexers.Get(strings.ToLower(k)) {
					continue
				}
				seen["f"+k] = true
				if got := lexers.Get(k); got != nil {
					rows = append(rows, kv{k, "f", got.Config().Name})
				}
			}
		}
		sort.Slice(rows, func(i, j int) bool { return rows[i].kind+rows[i].k < rows[j].kind+rows[j].k })
		for _, r := range rows {
			fmt.Printf("%s\t%s\t%s\n", r.kind, r.k, r.v)
		}
	case "tokens", "html":
		var items []item
		if err := json.NewDecoder(os.Stdin).Decode(&items); err != nil {
			panic(err)
		}
		res := make([]out, 0, len(items))
		for _, it := range items {
			o := out{Key: it.Key}
			if os.Args[1] == "tokens" {
				l := chromalexers.Get(it.Lang)
				if l != nil {
					o.Lexer = l.Config().Name
					iter, err := chroma.Coalesce(l).Tokenise(nil, it.Code)
					if err != nil {
						o.Err = err.Error()
					} else {
						var runs [][]any
						for _, tok := range iter.Tokens() {
							c := class(tok.Type)
							n := len([]rune(tok.Value))
							if len(runs) > 0 && runs[len(runs)-1][0] == c {
								runs[len(runs)-1][1] = runs[len(runs)-1][1].(int) + n
							} else {
								runs = append(runs, []any{c, n})
							}
						}
						o.Runs = runs
					}
				}
			} else {
				cfg := highlight.DefaultConfig
				if it.NoClasses != nil {
					cfg.NoClasses = *it.NoClasses
				}
				if it.Style != "" {
					cfg.Style = it.Style
				}
				h := highlight.New(cfg)
				s, err := h.Highlight(it.Code, it.Lang, it.Opts)
				o.HTML = s
				if err != nil {
					o.Err = err.Error()
				}
			}
			res = append(res, o)
		}
		enc := json.NewEncoder(os.Stdout)
		enc.SetEscapeHTML(false)
		if err := enc.Encode(res); err != nil {
			panic(err)
		}
	}
}
